9pub enum Direction {
10    /// Data received from a remote peer
11    In,
12    /// Data sent to a remote peer
13    Out,
14}
15
16impl Direction {
17    #[inline]
18    fn to_str(self) -> &'static str {
19        match self {
20            Self::In => "in",
21            Self::Out => "out",
22        }
23    }
24}

675#[inline]
676fn dir_to_string(dir: quinn::Dir) -> &'static str {
677    match dir {
678        quinn::Dir::Uni => "uni",
679        quinn::Dir::Bi => "bi",
680    }

683#[inline]
684fn connection_error_to_string(error: &quinn::ConnectionError) -> &'static str {
685    #[allow(clippy::enum_glob_use)]
686    use quinn::ConnectionError::*;
687
688    match error {
689        VersionMismatch => "version_mismatch",
690        TransportError(_) => "transport_error",
691        ConnectionClosed(_) => "connection_closed",
692        ApplicationClosed(_) => "application_closed",
693        Reset => "reset",
694        TimedOut => "timed_out",
695        LocallyClosed => "locally_closed",
696        CidsExhausted => "cids_exhausted",
697    }
