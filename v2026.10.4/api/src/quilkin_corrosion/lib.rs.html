26#[derive(Clone)]
27pub struct Clock(pub std::sync::Arc<uhlc::HLC>);
28
29impl Clock {
30    #[inline]
31    pub fn update_with_timestamp(
32        &self,
33        actor_id: types::actor::ActorId,
34        ts: types::broadcast::Timestamp,
35    ) -> Result<(), String> {
36        // The try_into().unwrap() is an unfortunate necessity, the uhlc lib only provides TryFrom implementations,
37        // but the conversion literally cannot fail, it would only fail if corrosion actorid increased in size beyond
38        // its current 16 bytes
39        self.0
40            .update_with_timestamp(&uhlc::Timestamp::new(
41                ts.to_ntp64(),
42                actor_id.try_into().unwrap(),
43            ))
44            .map_err(|e| e.to_string())
45    }
46
47    #[inline]
48    pub fn new_timestamp(&self) -> types::broadcast::Timestamp {
49        self.0.new_timestamp().into()
50    }
51}
