17use std::sync::atomic::AtomicBool;
18
19use std::panic;
20use std::sync::Arc;
21use std::sync::atomic::Ordering::Relaxed;
22
23#[derive(Clone)]
24pub struct Health {
25    healthy: Arc<AtomicBool>,
26    shutdown_tx: crate::signal::ShutdownTx,
27}
28
29impl Health {
30    pub fn new(shutdown_tx: crate::signal::ShutdownTx) -> Self {
31        let health = Self {
32            healthy: Arc::new(AtomicBool::new(true)),
33            shutdown_tx,
34        };
35
36        let healthy = health.healthy.clone();
37        let shutdown_tx = health.shutdown_tx.clone();
38        let default_hook = panic::take_hook();
39        panic::set_hook(Box::new(move |panic_info| {
40            tracing::error!(%panic_info, "Panic has occurred. Moving to Unhealthy");
41            healthy.swap(false, Relaxed);
42            let _ = shutdown_tx.send(());
43            default_hook(panic_info);
44        }));
45
46        health
47    }
48
49    /// returns a HTTP 200 response if the proxy is healthy.
50    pub fn check_liveness(&self) -> bool {
51        self.healthy.load(Relaxed)
52    }
53
54    /// State that the panic hook doesn't write to. The hook is process wide and
55    /// every instance chains onto it, so any test that panics marks every other
56    /// instance unhealthy too.
57    #[cfg(test)]
58    pub(super) fn detached(shutdown_tx: crate::signal::ShutdownTx) -> Self {
59        Self {
60            healthy: Arc::new(AtomicBool::new(true)),
61            shutdown_tx,
62        }
63    }
64
65    #[cfg(test)]
66    pub(super) fn mark_unhealthy(&self) {
67        self.healthy.store(false, Relaxed);
68    }
69}
70
71#[cfg(test)]
72mod tests {
73    use super::*;
74
75    #[test]
76    fn panic_hook() {
77        let (shutdown_tx, _shutdown_rx) = crate::signal::channel();
78        let health = Health::new(shutdown_tx);
79
80        // Deliberately not asserting liveness first: another test panicking
81        // would have marked this instance unhealthy already.
82        let _unused = std::panic::catch_unwind(|| {
83            panic!("oh no!");
84        });
85
86        assert!(!health.check_liveness());
87    }
88}
