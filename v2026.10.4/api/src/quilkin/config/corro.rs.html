20impl Drop for BridgeInner {
21    fn drop(&mut self) {
22        let statements = std::mem::take(&mut self.statements);
23
24        if !statements.is_empty() {
25            let count = statements.len();
26
27            if self.tx.tx.send(statements).is_ok() {
28                tracing::debug!(count, "sent statements from bridge");
29            } else {
30                use std::sync::atomic;
31
32                static OUTPUT: atomic::AtomicBool = atomic::AtomicBool::new(false);
33
34                if OUTPUT.load(atomic::Ordering::Relaxed) {
35                    return;
36                }
37
38                tracing::error!("xds bridge can no longer send updates, the receiver was closed");
39                OUTPUT.store(true, atomic::Ordering::Relaxed);
40            }
41        } else {
42            tracing::trace!("no statements to send");
43        }
44    }
45}

47pub struct Bridge<'m> {
48    maps: Option<(
49        super::watch::WatchGuard<'m, super::DatacenterMap>,
50        super::watch::WatchGuard<'m, crate::net::ClusterMap>,
51    )>,
52    b: Option<BridgeInner>,
53    applied: Applied,
54}
55
56impl<'m> Bridge<'m> {
57    pub fn new(config: &'m super::Config) -> Self {
58        Self {
59            maps: config
60                .dyn_cfg
61                .datacenters()
62                .map(|dc| dc.write())
63                .zip(config.dyn_cfg.clusters().map(|c| c.write())),
64            b: BridgeInner::new(config),
65            applied: Applied::All,
66        }
67    }
68
69    /// How much of what the bridge was given it took on.
70    pub fn applied(&self) -> Applied {
71        self.applied
72    }
73
74    pub fn insert_dc(&mut self, ip: IpAddr, dc: super::Datacenter) {
75        let Some((wdc, wsrv)) = &self.maps else {
76            return;
77        };
78
79        wdc.insert(ip, dc);
80
81        let Some(bri) = &mut self.b else {
82            return;
83        };
84
85        let peer = corrosion::ip_to_peer(ip);
86
87        let mut sv = corrosion::SmallVec::<[_; 100]>::new();
88        {
89            let mut d = corrosion::db::write::Datacenter(&mut sv);
90            d.insert(peer, dc.qcmp_port, dc.icao_code);
91        }
92
93        // We may have inserted clusters already from this DC, if so, insert all of them
94        if let Some(locality) = wsrv.locality_for_ip(ip)
95            && let Some(endpoints) = wsrv.get(&Some(locality))
96        {
97            let mut s = corrosion::db::write::Server::for_peer(peer, &mut sv);
98
99            for ep in endpoints.value().endpoint_iter() {
100                s.upsert(
101                    &quilkin_types::Endpoint {
102                        address: ep.address.host.clone(),
103                        port: ep.address.port,
104                    },
105                    dc.icao_code,
106                    &ep.metadata.known.tokens,
107                );
108            }
109        }
110
111        bri.statements.extend(sv);
112    }
113
114    pub fn remove_dc(&mut self, ip: IpAddr) {
115        let Some((wdc, _wsrv)) = &self.maps else {
116            return;
117        };
118
119        wdc.remove(ip);
120
121        let Some(bri) = &mut self.b else {
122            return;
123        };
124
125        let peer = corrosion::ip_to_peer(ip);
126
127        let mut sv = corrosion::SmallVec::<[_; 2]>::new();
128        {
129            let mut d = corrosion::db::write::Datacenter(&mut sv);
130            d.remove(peer, None);
131        }
132
133        bri.statements.extend(sv);
134    }
135
136    /// Drops an agent whose stream closed, in memory and in the database
137    ///
138    /// The servers it fed stay routable until the reaper collects them, giving
139    /// a restarting agent time to renew them
140    pub fn remove_agent(&mut self, ip: IpAddr) {
141        // The janitor may have dropped the entry already; the row outlives it
142        self.remove_dc(ip);
143
144        let Some((_wdc, wsrv)) = &self.maps else {
145            return;
146        };
147
148        wsrv.remove_contributor(Some(ip));
149    }
150
151    pub fn remove_locality(
152        &mut self,
153        ip: Option<IpAddr>,
154        locality: &Option<quilkin_xds::locality::Locality>,
155    ) {
156        let Some((_wdc, wsrv)) = &self.maps else {
157            return;
158        };
159
160        let Some(es) = wsrv.remove_locality(ip, locality) else {
161            return;
162        };
163
164        let Some((bri, ip)) = self.b.as_mut().zip(ip) else {
165            return;
166        };
167
168        let peer = corrosion::ip_to_peer(ip);
169
170        let mut statements = corrosion::SmallVec::<[_; 100]>::new();
171        {
172            let mut stx = corrosion::db::write::Server::for_peer(peer, &mut statements);
173
174            for rm_srv in es.into_endpoints() {
175                stx.remove_deferred(&rm_srv);
176            }
177        }
178
179        bri.statements.extend(statements);
180    }
181
182    pub fn apply_changes(
183        &mut self,
184        ip: Option<IpAddr>,
185        locality: Option<quilkin_xds::locality::Locality>,
186        icao: Option<quilkin_types::IcaoCode>,
187        es: crate::config::cluster::EndpointSet,
188    ) {
189        let Some((_wdc, wsrv)) = &self.maps else {
190            return;
191        };
192
193        let mut rm = Vec::new();
194        let mut up = Vec::new();
195
196        if let Err(error) = wsrv.apply(ip, locality, es, &mut rm, &mut up) {
197            // Held back rather than rejected: the sender can't do anything about
198            // it, and will be asked for these endpoints again once it can be
199            tracing::warn!(%error, "held back an endpoint update");
200            self.applied = Applied::Partial;
201            return;
202        }
203
204        let Some((bri, peer)) = self.b.as_mut().zip(ip.map(corrosion::ip_to_peer)) else {
205            return;
206        };
207
208        if icao.is_none() {
209            tracing::debug!(%peer, "applying endpoints for an unknown datacenter");
210        }
211
212        let mut statements = corrosion::SmallVec::<[_; 100]>::new();
213        {
214            let mut stx = corrosion::db::write::Server::for_peer(peer, &mut statements);
215
216            for rm_srv in rm {
217                stx.remove_deferred(&rm_srv);
218            }
219
220            for (ep, ts) in up {
221                stx.upsert(&ep, icao, &ts);
222            }
223        }
224
225        bri.statements.extend(statements);
226    }
