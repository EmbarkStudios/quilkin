58/// Installs the rustls crypto provider the TLS paths need.
59///
60/// `quilkin` does this in `main`, which no test runs. Tests in a binary share
61/// the process, so only the first call installs and the rest are no-ops.
62pub fn install_crypto_provider() {
63    static INSTALLED: std::sync::Once = std::sync::Once::new();
64    INSTALLED.call_once(|| {
65        drop(rustls::crypto::ring::default_provider().install_default());
66    });
67}

69/// A CA and the server certificate it signed, for a relay serving TLS.
70///
71/// The leaf is signed by a separate CA rather than being its own anchor:
72/// rustls rejects a self-signed leaf offered as its own trust root.
73pub struct TestPki {
74    pub ca_pem: String,
75    pub cert_pem: String,
76    pub key_pem: String,
77}

79impl TestPki {
80    pub fn generate() -> Self {
81        use rcgen::{
82            BasicConstraints, CertificateParams, DnType, ExtendedKeyUsagePurpose, IsCa, Issuer,
83            KeyPair, KeyUsagePurpose,
84        };
85
86        let mut ca_params = CertificateParams::new(Vec::new()).unwrap();
87        ca_params
88            .distinguished_name
89            .push(DnType::CommonName, "quilkin test ca");
90        ca_params.is_ca = IsCa::Ca(BasicConstraints::Unconstrained);
91        ca_params.key_usages = vec![KeyUsagePurpose::KeyCertSign, KeyUsagePurpose::CrlSign];
92        let ca_key = KeyPair::generate().unwrap();
93        let ca_cert = ca_params.self_signed(&ca_key).unwrap();
94
95        // The pails address the relay as `localhost`, so that is what the
96        // certificate has to be valid for.
97        let mut params = CertificateParams::new(vec!["localhost".to_owned()]).unwrap();
98        params
99            .distinguished_name
100            .push(DnType::CommonName, "localhost");
101        params.extended_key_usages = vec![ExtendedKeyUsagePurpose::ServerAuth];
102        let key = KeyPair::generate().unwrap();
103        let issuer = Issuer::from_params(&ca_params, &ca_key);
104        let cert = params.signed_by(&key, &issuer).unwrap();
105
106        Self {
107            ca_pem: ca_cert.pem(),
108            cert_pem: cert.pem(),
109            key_pem: key.serialize_pem(),
110        }
111    }
112}

114pub fn init_logging(level: Level, test_pkg: &'static str) {
115    use tracing_subscriber::{Layer as _, layer::SubscriberExt as _};
116    let layer = tracing_subscriber::fmt::layer()
117        .with_test_writer()
118        .with_filter(tracing_subscriber::filter::LevelFilter::from_level(level))
119        .with_filter(tracing_subscriber::EnvFilter::new(format!(
120            // NOTE: corrosion is _EXTREMELY_ spammy at the trace level, so don't enable these unless you are trying
121            // to see everything that could go on, but it's so noisy it can actually make it harder to diagnose issues
122            // corrosion=trace,corro_types=trace,tripwire=trace
123            "{test_pkg}=trace,qt=trace,quilkin=debug"
124        )));
125    let sub = tracing_subscriber::Registry::default().with(layer);
126    let disp = tracing::dispatcher::Dispatch::new(sub);
127    // tests in the same binary share the process-wide dispatcher, so only the
128    // first one gets to install it
129    drop(tracing::dispatcher::set_global_default(disp));
130}

340impl Pail {
341    #[inline]
342    pub fn config(&self) -> Arc<Config> {
343        match self {
344            Self::Relay(p) => p.config.clone(),
345            Self::Agent(p) => p.config.clone(),
346            Self::Proxy(p) => p.config.clone(),
347            Self::Server(_) => panic!("no config"),
348        }
349    }
350
351    pub fn is_healthy(&self) -> bool {
352        match self {
353            Pail::Relay(p) => p.healthy.load(std::sync::atomic::Ordering::Relaxed),
354            Pail::Agent(p) => p.healthy.load(std::sync::atomic::Ordering::Relaxed),
355            Pail::Proxy(p) => p.healthy.load(std::sync::atomic::Ordering::Relaxed),
356            Pail::Server(_) => panic!("no health"),
357        }
358    }
359
360    async fn stop(&mut self) {
361        if let Self::Relay(rp) = self {
362            if let Some(task) = rp.task.take() {
363                let _ = rp.shutdown.send(());
364                let result = task.await.unwrap();
365                tracing::info!(result = ?result.1, "task finished");
366            }
367        } else {
368            unimplemented!();
369        }
370    }
371
372    async fn start(&mut self) {
373        if let Self::Relay(rp) = self {
374            if rp.task.is_some() {
375                panic!("relay still running");
376            }
377
378            let svc = quilkin::Service::default()
379                .xds()
380                .xds_port(rp.xds_port)
381                .mds()
382                .mds_port(rp.mds_port)
383                .grpc()
384                .corrosion_port(0)
385                .testing();
386            let (tx, rx) = quilkin::signal::channel();
387            let sh = quilkin::signal::ShutdownHandler::new(tx.clone(), rx);
388            let (task, ports) = svc.spawn_services(&rp.config, sh).await.unwrap();
389
390            rp.shutdown = tx;
391            rp.corrosion_port = ports.corrosion;
392            rp.task = Some(task);
393        } else {
394            unimplemented!();
395        }
396    }
397}

399impl Pail {
400    pub async fn construct(spc: SandboxPailConfig, pails: &Pails, td: &std::path::Path) -> Self {
401        match spc.config {
402            PailConfig::Server(sspc) => {
403                let (packet_tx, packet_rx) = mpsc::channel::<String>(10);
404                let socket = quilkin::net::DualStackEpollSocket::new(0)
405                    .expect("failed to create server socket");
406
407                let port = socket
408                    .local_addr()
409                    .expect("failed to bind server socket")
410                    .port();
411
412                tracing::debug!(port, spc.name, "bound server socket");
413
414                let packet_size = sspc.packet_size as usize;
415                let mut num_packets = sspc.num_packets.unwrap_or(usize::MAX);
416
417                let task = tokio::spawn(async move {
418                    let mut buf = vec![0; packet_size];
419                    let mut received = 0;
420
421                    while num_packets > 0 {
422                        let (size, addr) = socket
423                            .recv_from(&mut buf)
424                            .await
425                            .expect("failed to receive packet");
426                        received += size;
427
428                        if sspc.echo {
429                            socket
430                                .send_to(&buf[..size], addr)
431                                .await
432                                .expect("failed to send packet");
433                        } else {
434                            let pstr = std::str::from_utf8(&buf[..size])
435                                .expect("received non-utf8 string in packet")
436                                .to_owned();
437
438                            packet_tx.send(pstr).await.expect("packet receiver dropped");
439                        }
440
441                        num_packets -= 1;
442                    }
443
444                    received
445                });
446
447                Self::Server(ServerPail {
448                    port,
449                    task: Some(task),
450                    packet_rx: Some(packet_rx),
451                })
452            }
453            PailConfig::Relay(rpc) => {
454                let path = td.join(spc.name);
455                let mut tc = rpc.config.unwrap_or_default();
456                tc.id = spc.name.into();
457                tc.write_to_file(&path);
458
459                let config_path = path.clone();
460
461                let (shutdown, shutdown_rx) = quilkin::signal::channel();
462                let pail_token = quilkin::signal::cancellation_token(shutdown_rx.clone());
463
464                let tls_ca_path = rpc.tls.then(|| {
465                    install_crypto_provider();
466                    let pki = TestPki::generate();
467                    let ca_path = td.join(format!("{}-ca.pem", spc.name));
468                    std::fs::write(&ca_path, &pki.ca_pem).unwrap();
469                    (ca_path, pki)
470                });
471
472                let providers = quilkin::Providers::default().fs().fs_path(path);
473                let mut svc = quilkin::Service::default()
474                    .xds()
475                    .xds_port(0)
476                    .mds()
477                    .mds_port(0)
478                    .grpc()
479                    .corrosion_port(0)
480                    .testing();
481
482                if let Some((_, pki)) = &tls_ca_path {
483                    svc = svc.tls_pem(&pki.cert_pem, &pki.key_pem);
484                }
485
486                let config = crate::Config::new_rc(
487                    Some("test-relay".into()),
488                    Default::default(),
489                    &providers,
490                    &mut svc,
491                    pail_token,
492                );
493
494                *config.dyn_cfg.id.lock() = spc.name.into();
495                let healthy = Arc::new(std::sync::atomic::AtomicBool::new(false));
496                let provider_task = providers
497                    .spawn_providers(&config, healthy.clone(), None, None, shutdown_rx.clone())
498                    .expect("failed to configure providers");
499                let (task, ports) = svc
500                    .spawn_services(
501                        &config,
502                        quilkin::signal::ShutdownHandler::new(shutdown.clone(), shutdown_rx),
503                    )
504                    .await
505                    .unwrap();
506
507                Self::Relay(RelayPail {
508                    xds_port: ports.xds.expect("xds not spawned"),
509                    mds_port: ports.mds.expect("mds not spawned"),
510                    tls_ca_path: tls_ca_path.map(|(path, _)| path),
511                    corrosion_port: ports.corrosion,
512                    task: Some(task),
513                    provider_task,
514                    healthy,
515                    shutdown,
516                    config_file: Some(ConfigFile {
517                        path: config_path,
518                        config: tc,
519                    }),
520                    config,
521                })
522            }
523            PailConfig::Agent(apc) => {
524                let mut endpoints = std::collections::BTreeSet::new();
525
526                for (dep_name, tokens) in apc.endpoints {
527                    let Pail::Server(server) = &pails[dep_name] else {
528                        panic!("expected '{dep_name}' to be a server pail for endpoint");
529                    };
530
531                    endpoints.insert(quilkin::net::Endpoint::with_metadata(
532                        (std::net::Ipv4Addr::UNSPECIFIED, server.port).into(),
533                        quilkin::net::endpoint::Metadata {
534                            tokens: tokens.iter().map(|t| Vec::from(*t)).collect(),
535                        },
536                    ));
537                }
538
539                let mut tc = TestConfig::new();
540                tc.clusters.insert_default(endpoints);
541                tc.id = spc.name.into();
542
543                let path = td.join(spc.name);
544                tc.write_to_file(&path);
545
546                let mut relay_ca = None;
547                let relay_servers = spc
548                    .dependencies
549                    .iter()
550                    .filter_map(|dname| {
551                        let Pail::Relay(RelayPail {
552                            mds_port,
553                            tls_ca_path,
554                            ..
555                        }) = &pails[dname]
556                        else {
557                            return None;
558                        };
559                        let scheme = if tls_ca_path.is_some() {
560                            relay_ca = tls_ca_path.clone();
561                            "https"
562                        } else {
563                            "http"
564                        };
565                        Some(
566                            format!("{scheme}://localhost:{mds_port}")
567                                .parse()
568                                .expect("failed to parse endpoint"),
569                        )
570                    })
571                    .collect::<Vec<_>>();
572
573                let (shutdown, shutdown_rx) = quilkin::signal::channel();
574                let pail_token = quilkin::signal::cancellation_token(shutdown_rx.clone());
575
576                let config_path = path.clone();
577                let mut svc = quilkin::Service::default().qcmp().qcmp_port(0);
578                let mut providers = quilkin::Providers::default()
579                    .fs()
580                    .fs_path(path)
581                    .grpc_push_endpoints(relay_servers);
582
583                if let Some(ca) = relay_ca {
584                    install_crypto_provider();
585                    providers = providers.grpc_push_tls_ca_path(ca);
586                }
587
588                let corrosion_eps = spc
589                    .dependencies
590                    .iter()
591                    .filter_map(|dname| {
592                        let Pail::Relay(RelayPail { corrosion_port, .. }) = &pails[dname] else {
593                            return None;
594                        };
595                        corrosion_port.filter(|_a| apc.corrosion).map(|p| {
596                            quilkin::net::EndpointAddress::from(SocketAddr::V6(
597                                std::net::SocketAddrV6::new(std::net::Ipv6Addr::LOCALHOST, p, 0, 0),
598                            ))
599                        })
600                    })
601                    .collect::<Vec<_>>();
602
603                let providers = if !corrosion_eps.is_empty() {
604                    providers
605                        .corrosion_endpoints(corrosion_eps)
606                        .corrosion_mode(quilkin::providers::corrosion::CorrosionMode::Push)
607                } else {
608                    providers
609                };
610
611                let config = crate::Config::new_rc(
612                    Some("test-agent".into()),
613                    apc.icao_code,
614                    &providers,
615                    &mut svc,
616                    pail_token,
617                );
618
619                *config.dyn_cfg.id.lock() = spc.name.into();
620                let acfg = config.clone();
621                let healthy = Arc::new(std::sync::atomic::AtomicBool::new(false));
622                let provider_task = providers
623                    .spawn_providers(&config, healthy.clone(), None, None, shutdown_rx.clone())
624                    .expect("failed to configure providers");
625                let (task, ports) = svc
626                    .spawn_services(
627                        &config,
628                        quilkin::signal::ShutdownHandler::new(shutdown.clone(), shutdown_rx),
629                    )
630                    .await
631                    .unwrap();
632
633                Self::Agent(AgentPail {
634                    qcmp_port: ports.qcmp.expect("failed to spawn QCMP"),
635                    task: Some(task),
636                    provider_task,
637                    healthy,
638                    shutdown,
639                    config_file: Some(ConfigFile {
640                        path: config_path,
641                        config: tc,
642                    }),
643                    config: acfg,
644                })
645            }
646            PailConfig::Proxy(ppc) => {
647                let providers = if ppc.corrosion {
648                    let corrosion_eps: Vec<_> = spc
649                        .dependencies
650                        .iter()
651                        .filter_map(|dname| {
652                            let Pail::Relay(RelayPail { corrosion_port, .. }) = &pails[dname]
653                            else {
654                                return None;
655                            };
656                            corrosion_port.map(|p| {
657                                quilkin::net::EndpointAddress::from(SocketAddr::V6(
658                                    std::net::SocketAddrV6::new(
659                                        std::net::Ipv6Addr::LOCALHOST,
660                                        p,
661                                        0,
662                                        0,
663                                    ),
664                                ))
665                            })
666                        })
667                        .collect();
668
669                    quilkin::Providers::default()
670                        .corrosion_endpoints(corrosion_eps)
671                        .corrosion_mode(quilkin::providers::corrosion::CorrosionMode::Pull)
672                } else {
673                    let mut relay_ca = None;
674                    let management_servers: Vec<_> = spc
675                        .dependencies
676                        .iter()
677                        .filter_map(|dname| {
678                            let Pail::Relay(RelayPail {
679                                xds_port,
680                                tls_ca_path,
681                                ..
682                            }) = &pails[dname]
683                            else {
684                                return None;
685                            };
686                            let scheme = if tls_ca_path.is_some() {
687                                relay_ca = tls_ca_path.clone();
688                                "https"
689                            } else {
690                                "http"
691                            };
692                            Some(
693                                format!("{scheme}://localhost:{xds_port}")
694                                    .parse()
695                                    .expect("failed to parse endpoint"),
696                            )
697                        })
698                        .collect();
699
700                    let mut providers =
701                        quilkin::Providers::default().grpc_pull_endpoints(management_servers);
702
703                    if let Some(ca) = relay_ca {
704                        install_crypto_provider();
705
706                        // A CA that signed nothing in this sandbox, for asserting
707                        // that verification is actually performed.
708                        let ca = if ppc.mismatched_tls_ca {
709                            let path = td.join(format!("{}-wrong-ca.pem", spc.name));
710                            std::fs::write(&path, TestPki::generate().ca_pem).unwrap();
711                            path
712                        } else {
713                            ca
714                        };
715
716                        providers = providers.grpc_pull_tls_ca_path(ca);
717                    }
718
719                    providers
720                };
721
722                let mut svc = quilkin::Service::default()
723                    .udp()
724                    .udp_port(0)
725                    .qcmp()
726                    .qcmp_port(0)
727                    .phoenix()
728                    .phoenix_port(0)
729                    .termination_timeout(None);
730
731                let (shutdown, shutdown_rx) = quilkin::signal::channel();
732                let pail_token = quilkin::signal::cancellation_token(shutdown_rx.clone());
733
734                let config = crate::Config::new_rc(
735                    Some("test-proxy".into()),
736                    Default::default(),
737                    &Default::default(),
738                    &mut svc,
739                    pail_token,
740                );
741
742                if let Some(cfg) = ppc.config {
743                    if !cfg.clusters.is_empty() {
744                        panic!("not implemented");
745                    }
746
747                    if !cfg.filters.is_empty() {
748                        config.dyn_cfg.filters().unwrap().store(cfg.filters);
749                    }
750                }
751
752                let endpoints: std::collections::BTreeSet<_> = spc
753                    .dependencies
754                    .iter()
755                    .filter_map(|dname| {
756                        let Pail::Server(ServerPail { port, .. }) = &pails[dname] else {
757                            return None;
758                        };
759
760                        Some(quilkin::net::Endpoint::new(
761                            (std::net::Ipv6Addr::LOCALHOST, *port).into(),
762                        ))
763                    })
764                    .collect();
765
766                if !endpoints.is_empty() {
767                    config
768                        .dyn_cfg
769                        .clusters()
770                        .unwrap()
771                        .modify(|clusters| clusters.insert_default(endpoints));
772                }
773
774                *config.dyn_cfg.id.lock() = spc.name.into();
775
776                let (rttx, rtrx) = tokio::sync::mpsc::unbounded_channel();
777
778                let healthy = Arc::new(std::sync::atomic::AtomicBool::new(false));
779                let provider_task = providers
780                    .spawn_providers(
781                        &config,
782                        healthy.clone(),
783                        None,
784                        Some(rttx),
785                        shutdown_rx.clone(),
786                    )
787                    .expect("failed to configure providers");
788                let (task, ports) = svc
789                    .spawn_services(
790                        &config,
791                        quilkin::signal::ShutdownHandler::new(shutdown.clone(), shutdown_rx),
792                    )
793                    .await
794                    .unwrap();
795
796                Self::Proxy(ProxyPail {
797                    port: ports.udp.expect("failed to spawn proxy"),
798                    qcmp_port: ports.qcmp.expect("failed to spawn qcmp"),
799                    phoenix_port: ports.phoenix.expect("failed to spawn phoenix"),
800                    shutdown,
801                    task: Some(task),
802                    provider_task,
803                    healthy,
804                    config,
805                    delta_applies: Some(rtrx),
806                })
807            }
808        }
809    }
810}

818macro_rules! sandbox_config {
819    () => {{
820        let fname = $crate::func_name!();
821        let mut name = String::new();
822        for comp in fname.split("::") {
823            if comp.starts_with("{{") {
824                continue;
825            }
826
827            if !name.is_empty() {
828                name.push('.');
829            }
830
831            name.push_str(comp);
832        }
833        $crate::SandboxConfig::new(name)
834    }};
835}

837impl SandboxConfig {
838    pub fn new(name: impl Into<String>) -> Self {
839        quilkin_xds::metrics::set_registry(quilkin::metrics::registry());
840
841        Self {
842            name: name.into(),
843            pails: Vec::new(),
844        }
845    }
846
847    pub fn push(
848        &mut self,
849        name: &'static str,
850        config: impl Into<PailConfig>,
851        deps: &'static [&'static str],
852    ) {
853        self.pails.push(SandboxPailConfig {
854            name,
855            config: config.into(),
856            dependencies: deps,
857        });
858    }
859
860    pub async fn spinup(self) -> Sandbox {
861        // Validate that every dependency is satisfied
862        for i in 0..self.pails.len() {
863            let deps = self.pails[i].dependencies;
864
865            for dep in deps {
866                if !self.pails[..i].iter().any(|sp| sp.name == *dep) {
867                    panic!(
868                        "failed to locate dependency '{dep}' for '{}'",
869                        self.pails[i].name
870                    );
871                }
872            }
873        }
874
875        tracing::trace!("dependencies resolved");
876
877        // Create a channel that is used to communicate readiness of each pail we create
878        //let (tx, rx) = tokio::sync::mpsc::unbounded_channel();
879
880        // Not necessarily used in every test, but in most
881        let td = match tempfile::TempDir::with_prefix(&self.name) {
882            Ok(td) => td,
883            Err(err) => {
884                panic!("failed to create temp dir '{}': {err:#}", self.name);
885            }
886        };
887
888        let mut pails = Pails::new();
889        for pc in self.pails {
890            let name = pc.name;
891            let pail = Pail::construct(pc, &pails, td.path()).await;
892
893            if pails.insert(name, pail).is_some() {
894                panic!("{name} already existed");
895            }
896        }
897
898        Sandbox { pails, td }
899    }
900}

1052    async fn wait_for_health_state(&self, which: &str, health_state: bool) {
1053        let Some((_, pail)) = self.pails.iter().find(|(name, _)| **name == which) else {
1054            panic!("failed to find '{which}'");
1055        };
1056
1057        let start = tokio::time::Instant::now();
1058        while pail.is_healthy() != health_state {
1059            tracing::debug!(
1060                "waiting for '{}' to enter health state '{}'",
1061                which,
1062                health_state
1063            );
1064            tokio::time::sleep(std::time::Duration::from_secs(1)).await;
1065
1066            if start.elapsed() > MAX_WAIT {
1067                panic!(
1068                    "timed out waiting for '{which}' to enter healthy state '{health_state}' after {MAX_WAIT:?}"
1069                );
1070            }
1071        }

1074    pub async fn block_until_packet_gets_through(
1075        &self,
1076        msg: &[u8],
1077        expected: &str,
1078        client: &quilkin::net::DualStackEpollSocket,
1079        proxy_address: &std::net::SocketAddr,
1080        server_rx: &mut tokio::sync::mpsc::Receiver<String>,
1081    ) {
1082        const PACKET_TIMEOUT: std::time::Duration = std::time::Duration::from_millis(500);
1083        let start = tokio::time::Instant::now();
1084        loop {
1085            tracing::debug!(len = msg.len(), "sending packet");
1086            client.send_to(msg, *proxy_address).await.unwrap();
1087
1088            if let Some(rmsg) = self
1089                .maybe_timeout(PACKET_TIMEOUT.as_millis() as u64, server_rx.recv())
1090                .await
1091            {
1092                tracing::debug!(len = msg.len(), "received packet");
1093                match rmsg {
1094                    Some(rmsg) => {
1095                        assert_eq!(expected, rmsg);
1096                    }
1097                    None => {
1098                        panic!("got None message, was server_rx channel closed?");
1099                    }
1100                }
1101                break;
1102            }
1103            tracing::warn!(len = msg.len(), "packet timeout, probably dropped");
1104
1105            if start.elapsed() > MAX_WAIT {
1106                panic!("unable to pass message through proxy after {MAX_WAIT:?}");
1107            }
1108        }
1109    }
