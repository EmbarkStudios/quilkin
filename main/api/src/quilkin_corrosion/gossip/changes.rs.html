22/// State for dynamic batch processing
23struct HandleChangesState {
24    queue: VecDeque<(ChangeAndSource, Instant)>,
25    buf_cost: usize,
26    current_batch_size: usize,
27    processing_task: Option<tokio::task::JoinHandle<Result<(), ChangeError>>>,
28    max_wait: Option<std::pin::Pin<Box<tokio::time::Sleep>>>,
29
30    /// When emergency halving last occurred; prevents batch size re-inflation
31    /// during the grace period to avoid oscillation (halve → burst → halve …)
32    halved_at: Option<Instant>,
33
34    // Configuration
35    max_queue_len: usize,
36    min_batch_size: usize,
37    step_base: usize,
38    max_batch_size: usize,
39    batch_threshold_ratio: f64,
40    timeout_duration: Duration,
41
42    /// Duration after emergency halving during which batch size increases are blocked.
43    ///
44    /// Uses the transaction timeout as a reasonable proxy for "one full processing cycle".
45    tx_timeout: Duration,
46
47    metrics: &'static GossipMetrics,
48}
49
50impl HandleChangesState {
51    #[allow(clippy::too_many_arguments)]
52    fn new(
53        min_batch_size: usize,
54        step_base: usize,
55        max_batch_size: usize,
56        batch_threshold_ratio: f64,
57        timeout_duration: Duration,
58        tx_timeout: Duration,
59        max_queue_len: usize,
60        metrics: &'static GossipMetrics,
61    ) -> Self {
62        Self {
63            queue: VecDeque::with_capacity(max_queue_len),
64            buf_cost: 0,
65            current_batch_size: min_batch_size,
66            processing_task: None,
67            max_wait: Some(Box::pin(tokio::time::sleep(timeout_duration))),
68            halved_at: None,
69            min_batch_size,
70            step_base,
71            max_batch_size,
72            batch_threshold_ratio,
73            timeout_duration,
74            tx_timeout,
75            max_queue_len,
76            metrics,
77        }
78    }
79
80    /// Whether we're still in the grace period after an emergency halving.
81    #[inline]
82    fn in_grace_period(&self) -> bool {
83        self.halved_at
84            .is_some_and(|t| t.elapsed() < self.tx_timeout)
85    }
86
87    /// Calculate exponential batch size based on cost.
88    ///
89    /// During the grace period after emergency halving, the result is clamped
90    /// to `current_batch_size` to prevent re-inflation and oscillation.
91    #[inline]
92    fn calculate_batch_size(&self, cost: usize) -> usize {
93        let computed = if self.step_base == 0 || cost < self.step_base {
94            self.min_batch_size
95        } else {
96            let size = self.step_base * (1 << (cost / self.step_base).ilog2());
97            size.clamp(self.min_batch_size, self.max_batch_size)
98        };
99        if self.in_grace_period() {
100            computed.min(self.current_batch_size)
101        } else {
102            computed
103        }
104    }
105
106    /// Get the threshold for immediate spawning based on configured ratio
107    #[inline]
108    fn batch_threshold(&self) -> usize {
109        (self.current_batch_size as f64 * self.batch_threshold_ratio) as usize
110    }
111
112    /// Drain a batch from the queue and spawn processing task
113    fn drain_and_spawn(
114        &mut self,
115        ctx: &ChangeCtx,
116        target_size: usize,
117        reason: &'static str,
118    ) -> usize {
119        // Must complete previous task before spawning a new one
120        if self.processing_task.is_some() {
121            unreachable!("a processing task already exists");
122        }
123
124        let mut batch = Vec::with_capacity(self.current_batch_size);
125        let mut batch_cost = 0;
126
127        while let Some((change, queued_at)) = self.queue.pop_front() {
128            let cost = change.0.processing_cost();
129            batch_cost += cost;
130            self.buf_cost -= cost;
131            batch.push((change, queued_at));
132
133            if batch_cost >= target_size {
134                break;
135            }
136        }
137
138        if batch.is_empty() {
139            unreachable!("batch is empty");
140        }
141
142        let ctx = ctx.clone();
143        self.processing_task = Some(tokio::spawn(process_multiple_changes(
144            ctx,
145            batch,
146            self.tx_timeout,
147        )));
148
149        self.metrics.change_batches_inc(reason);
150        self.metrics.change_batch_size(self.current_batch_size);
151
152        batch_cost
153    }
154
155    /// Handle task completion and decide whether to spawn next batch
156    fn handle_task_completion(
157        &mut self,
158        ctx: &ChangeCtx,
159        result: Result<Result<(), corro_types::agent::ChangeError>, tokio::task::JoinError>,
160    ) {
161        // Handle task result
162        match result {
163            Ok(Ok(())) => {}
164            Ok(Err(error)) => {
165                tracing::error!(%error, "error processing change batch");
166
167                if let Some(issue) = error.fatal_db_issue() {
168                    super::fatal_db_issue(issue, &ctx.shutdown);
169                }
170
171                // Check for memory errors and emergency reduce batch size
172                // TODO: requeue the changes
173                if error.is_oom_error() || error.is_interrupt_error() {
174                    if self.current_batch_size == self.min_batch_size {
175                        tracing::error!(current_batch_size = %self.current_batch_size, min_batch_size = %self.min_batch_size, "batch too large for the database to process, but already at min_batch_size — min_batch_size may be too large or transaction timeout may be misconfigured");
176                    } else {
177                        tracing::error!(current_batch_size = %self.current_batch_size, "batch too large for the database to process, halving batch size");
178                    }
179                    self.current_batch_size =
180                        (self.current_batch_size / 2).max(self.min_batch_size);
181                    self.halved_at = Some(std::time::Instant::now());
182                }
183            }
184            Err(error) => {
185                tracing::error!(%error, "batch processing task panicked");
186            }
187        }
188
189        self.processing_task = None;
190
191        // Should we spawn immediately with queued changes?
192        if self.buf_cost >= self.batch_threshold() {
193            // YES - enough changes queued, spawn immediately
194            // Check if we might want to burst
195            if self.buf_cost > self.current_batch_size {
196                self.current_batch_size = self.calculate_batch_size(self.buf_cost);
197            }
198            self.drain_and_spawn(ctx, self.current_batch_size, "immediate-after-completion");
199        } else {
200            // NO - not enough changes yet, start timeout
201            self.max_wait = Some(Box::pin(tokio::time::sleep(self.timeout_duration)));
202        }
203    }
204
205    /// Handle timeout - spawn whatever we have
206    ///
207    /// The amount of changes MUST be smaller than the threshold
208    /// as otherwise other code paths would have spawned a batch
209    fn handle_timeout(&mut self, ctx: &ChangeCtx) {
210        self.metrics.changes_queued(self.buf_cost, self.queue.len());
211
212        self.max_wait = None;
213
214        if !self.queue.is_empty() {
215            // Just process whatever we have and then downsize the batch size
216            self.drain_and_spawn(ctx, usize::MAX, "timeout");
217
218            // Adjust batch size based on what we had processed
219            // This will decrease the batch size unless it's already at the minimum
220            self.current_batch_size = self.calculate_batch_size(self.buf_cost);
221        } else if self.processing_task.is_none() && self.max_wait.is_none() {
222            // Empty queue, no task running, no timeout set
223            self.current_batch_size = self.calculate_batch_size(0);
224            // If no task is running and no timeout is set, start timeout
225            // This ensures changes get processed even if they don't hit the threshold
226            self.max_wait = Some(Box::pin(tokio::time::sleep(self.timeout_duration)));
227        }
228    }
229
230    // Drops oldest items when the queue is full
231    #[inline]
232    fn maybe_drop_old_change(&mut self) -> Option<ChangeV1> {
233        let maybe_dropped_change = if self.queue.len() >= self.max_queue_len {
234            if let Some((dropped_change, _)) = self.queue.pop_front() {
235                self.buf_cost -= dropped_change.0.processing_cost();
236                Some(dropped_change.0)
237            } else {
238                None
239            }
240        } else {
241            None
242        };
243
244        if maybe_dropped_change.is_some() {
245            self.metrics.change_dropped_inc();
246        }
247
248        maybe_dropped_change
249    }
250
251    /// Handle new change arrival - check if we should spawn immediately
252    fn handle_new_change(&mut self, ctx: &ChangeCtx, change: ChangeAndSource) {
253        let cost = change.0.processing_cost();
254        self.queue.push_back((change, Instant::now()));
255        self.buf_cost += cost;
256
257        // Check if we should spawn immediately (threshold-based)
258        if self.buf_cost < self.batch_threshold() || self.processing_task.is_some() {
259            return;
260        }
261
262        // Cancel any pending timeout
263        self.max_wait = None;
264
265        // Check if we might want to burst
266        if self.buf_cost > self.current_batch_size {
267            self.current_batch_size = self.calculate_batch_size(self.buf_cost);
268        }
269
270        // Drain and spawn
271        self.drain_and_spawn(ctx, self.current_batch_size, "size-threshold-on-new-change");
272    }
273}

276pub struct ChangeOptions {
277    /// How many unapplied changesets corrosion will buffer before starting to drop them
278    pub processing_queue_len: usize,
279    /// Minimum amount of changes corrosion will try to apply at once in the same transaction
280    pub apply_queue_min_batch_size: usize,
281    /// `batch_size = clamp(min_batch_size, step_base * 2 ** floor(log2(x/step_base)), max_batch_size)`
282    pub apply_queue_step_base: usize,
283    /// Maximum amount of changes corrosion will try to apply at once in the same transaction
284    pub apply_queue_max_batch_size: usize,
285    /// Threshold ratio (0.0-1.0) for immediate batch spawning when queue reaches this fraction of batch size
286    ///
287    /// It's used to decide whether to wait for more changes for `apply_queue_timeout` ms or spawn a batch immediately
288    pub apply_queue_batch_threshold_ratio: f64,
289    /// Amount of time corrosion will wait before proceeding to apply a batch of changes
290    ///
291    /// We wait either for `apply_queue_timeout` or until at least `apply_queue_min_batch_size` changes accumulate
292    pub apply_queue_timeout: Duration,
293    /// Max amount of a time allowed for an individual SQL transaction
294    pub sql_tx_timeout: Duration,
295}

338/// Bundle incoming changes to optimise transaction sizes with `SQLite`
339///
340/// *Performance tradeoff*: introduce latency (with a max timeout) to
341/// apply changes more efficiently.
342///
343/// This function used by broadcast receivers and sync receivers
344pub async fn handle_changes(
345    ctx: ChangeCtx,
346    opts: ChangeOptions,
347    mut changes_rx: mpsc::Receiver<super::Change>,
348    mut tripwire: tripwire::Tripwire,
349) {
350    // Initialize batch processor state
351    let mut state = HandleChangesState::new(
352        opts.apply_queue_min_batch_size,
353        opts.apply_queue_step_base,
354        opts.apply_queue_max_batch_size,
355        opts.apply_queue_batch_threshold_ratio,
356        opts.apply_queue_timeout,
357        opts.sql_tx_timeout,
358        opts.processing_queue_len,
359        ctx.metrics,
360    );
361
362    let max_seen_cache_len = opts.processing_queue_len;
363
364    // unlikely, but max_seen_cache_len can be less than 10, in that case we want to just clear the whole cache
365    let keep_seen_cache_size = if max_seen_cache_len > 10 {
366        10.max(max_seen_cache_len / 10)
367    } else {
368        0
369    };
370
371    let mut seen = indexmap::IndexMap::<_, RangeInclusiveSet<CrsqlSeq>>::new();
372
373    loop {
374        let changes = tokio::select! {
375            biased;
376
377            // Processing task finished
378            res = async { state.processing_task.as_mut().unwrap().await }, if state.processing_task.is_some() => {
379                state.handle_task_completion(&ctx, res);
380                continue;
381            },
382
383            // New changes arrive
384            maybe_change_src = changes_rx.recv() => match maybe_change_src {
385                Some(change) => change,
386                None => break,
387            },
388
389            // Timeout fires
390            _ = async { state.max_wait.as_mut().unwrap().await }, if state.max_wait.is_some() => {
391                state.handle_timeout(&ctx);
392
393                // Cleanup seen cache
394                if seen.len() > max_seen_cache_len {
395                    seen.drain(..seen.len() - keep_seen_cache_size);
396                }
397
398                continue;
399            },
400
401            // Corrosion is shutting down
402            _ = &mut tripwire => {
403                break;
404            }
405        };
406
407        ctx.metrics.change_received_inc(changes.len());
408
409        for (change, src) in changes {
410            // Skip changes from ourselves
411            // TODO: seems better to just ensure we don't send to ourselves?
412            if change.actor_id == ctx.actor_id {
413                continue;
414            }
415
416            // Skip changes we've already seen recently in the seen cache
417            if let Some(mut seqs) = change.seqs() {
418                let v = change.versions().start();
419                if let Some(seen_seqs) = seen.get(&(change.actor_id, v))
420                    && seqs.all(|seq| seen_seqs.contains(&seq))
421                {
422                    continue;
423                }
424            } else if change
425                .versions()
426                .all(|v| seen.contains_key(&(change.actor_id, v)))
427            {
428                if src.is_broadcast() {
429                    ctx.metrics.change_broadcast_duplicate_inc("cache");
430                }
431                continue;
432            }
433
434            let src_str = src.as_str();
435
436            // Update logical clock if needed
437            let recv_lag = change.ts().and_then(|ts| {
438                let mut our_ts = ctx.clock.new_timestamp();
439                if ts > our_ts {
440                    if let Err(error) = ctx.clock.update_with_timestamp(change.actor_id, ts) {
441                        tracing::error!(remote_actor_id = %change.actor_id, error, "could not update clock from actor");
442                        return None;
443                    }
444
445                    ctx.metrics.change_clock_inc(src_str);
446
447                    // update our_ts to the new timestamp
448                    our_ts = ctx.clock.new_timestamp();
449                }
450                Some((our_ts.0 - ts.0).to_duration())
451            });
452
453            if src.is_broadcast() {
454                ctx.metrics.change_broadcast_received_inc("change");
455            }
456
457            // Skip changes we've already seen in the bookie
458            if let Some(booked) = ctx.bookie.get(&change.actor_id)
459                && booked.read().contains_all(change.versions(), change.seqs())
460            {
461                if src.is_broadcast() {
462                    ctx.metrics.change_broadcast_duplicate_inc("bookie");
463                }
464                continue;
465            }
466
467            if let Some(recv_lag) = recv_lag {
468                ctx.metrics
469                    .change_recv_lag_sample(recv_lag.as_secs_f64(), src_str);
470            }
471
472            // If we need to drop an old change, drop it from the seen cache as well
473            while let Some(dropped_change) = state.maybe_drop_old_change() {
474                for v in dropped_change.versions() {
475                    let indexmap::map::Entry::Occupied(mut entry) =
476                        seen.entry((change.actor_id, v))
477                    else {
478                        continue;
479                    };
480
481                    if let Some(seqs) = dropped_change.seqs() {
482                        entry.get_mut().remove(seqs.into());
483                    } else {
484                        entry.swap_remove_entry();
485                    }
486                }
487            }
488
489            // Register the new change in the seen cache
490            // this will only run once for a non-empty changeset
491            for v in change.versions() {
492                let entry = seen.entry((change.actor_id, v)).or_default();
493                if let Some(seqs) = change.seqs() {
494                    entry.extend([seqs.into()]);
495                }
496            }
497
498            // Rebroadcast changes received from broadcast
499            if src.is_broadcast()
500                && !change.is_empty()
501                && ctx
502                    .bcast_tx
503                    .try_send(bx::BroadcastInput::Rebroadcast(bx::BroadcastV1::Change(
504                        change.clone(),
505                    )))
506                    .is_err()
507            {
508                tracing::debug!("broadcasts are full or done!");
509            }
510
511            // Handle the new change - queue it and potentially spawn a batch
512            state.handle_new_change(&ctx, (change, src));
513        }
514    }
515}

517async fn process_multiple_changes(
518    ctx: ChangeCtx,
519    changes: Vec<(ChangeAndSource, Instant)>,
520    tx_timeout: Duration,
521) -> Result<(), corro_types::agent::ChangeError> {
522    let start = Instant::now();
523    //counter!("corro.agent.changes.processing.started").increment(changes.len() as u64);
524
525    const PROCESSING_WARN_THRESHOLD: Duration = Duration::from_secs(5);
526
527    let mut seen = HashSet::new();
528    let mut unknown_changes = BTreeMap::<ActorId, (Booked, Vec<_>)>::new();
529
530    for ((change, src), queued_at) in changes {
531        ctx.metrics.changes_queued_time(queued_at.elapsed());
532
533        let versions = change.versions();
534        let seqs = change.seqs();
535
536        if !seen.insert((change.actor_id, versions, seqs)) {
537            continue;
538        }
539
540        let booked = ctx.bookie.ensure(change.actor_id);
541        if booked.read().contains_all(change.versions(), change.seqs()) {
542            continue;
543        }
544
545        unknown_changes
546            .entry(change.actor_id)
547            .and_modify(|(_, per_actor_changes)| per_actor_changes.push((change.clone(), src)))
548            .or_insert_with(|| (booked, vec![(change, src)]));
549    }
550
551    let elapsed = start.elapsed();
552    if elapsed >= PROCESSING_WARN_THRESHOLD {
553        tracing::warn!(
554            ?elapsed,
555            "process_multiple_changes: removing duplicates took too long"
556        );
557    }
558
559    if unknown_changes.is_empty() {
560        return Ok(());
561    }
562
563    let mut conn = ctx.pool.write_normal().await?;
564    let ctx = ctx.clone();
565
566    let changesets = tokio::task::block_in_place(move || {
567        let bookie_write = ctx.bookie.write_lock_blocking();
568
569        let tx = conn
570            .immediate_transaction()
571            .map_err(|source| ChangeError::Rusqlite {
572                source,
573                actor_id: None,
574                version: None,
575            })?;
576
577        let mut tx = sqlite_pool::InterruptibleTransaction::new(
578            tx,
579            Some(tx_timeout),
580            "process_multiple_changes",
581        );
582
583        let mut processed = BTreeMap::<ActorId, Vec<_>>::new();
584        let mut changesets = Vec::new();
585        let mut booked_writes = BTreeMap::new();
586
587        for (actor_id, (booked, _)) in &unknown_changes {
588            booked_writes.insert(*actor_id, bookie_write.write_tx(booked));
589        }
590
591        let sub_start = Instant::now();
592        for (actor_id, (_, changes)) in unknown_changes {
593            let booked_write = booked_writes
594                .get_mut(&actor_id)
595                .expect("booked write guard should be present for every actor");
596
597            let max = booked_write.last();
598            let mut seen = rangemap::RangeInclusiveMap::new();
599
600            for (change, src) in changes {
601                let seqs = change.seqs();
602                if booked_write.contains_all(change.versions(), change.seqs()) {
603                    continue;
604                }
605
606                let versions = change.versions();
607
608                // check if we've seen this version here...
609                if versions.clone().all(|version| match seqs {
610                    Some(mut check_seqs) => match seen.get(&version) {
611                        Some(maybe_partial) => match maybe_partial {
612                            Some(agent::PartialVersion { seqs, .. }) => {
613                                check_seqs.all(|seq| seqs.contains(&seq))
614                            }
615                            // other kind of known version
616                            None => true,
617                        },
618                        None => false,
619                    },
620                    None => seen.contains_key(&version),
621                }) {
622                    continue;
623                }
624
625                // optimizing this, insert later!
626                let known = if change.is_complete() && change.is_empty() {
627                    let end = change.versions().end();
628
629                    // update db_version in db if it's greater than the max
630                    // since we aren't passing any changes to crsql
631                    if max.is_none_or(|max| end > max) {
632                        drop(
633                            tx.prepare_cached("SELECT crsql_set_db_version(?, ?)")
634                                .map_err(|source| ChangeError::Rusqlite {
635                                    source,
636                                    actor_id: Some(change.actor_id),
637                                    version: Some(end),
638                                })?
639                                .query_row((change.actor_id, end), |row| row.get::<_, String>(0))
640                                .map_err(|source| ChangeError::Rusqlite {
641                                    source,
642                                    actor_id: Some(change.actor_id),
643                                    version: Some(end),
644                                })?,
645                        );
646                    }
647
648                    KnownDbVersion::Cleared
649                } else {
650                    if let Some(seqs) = change.seqs()
651                        && seqs.end() < seqs.start()
652                    {
653                        tracing::warn!(%actor_id, versions = ?change.versions(), "received an invalid change, seqs start is greater than seqs end: {seqs:?}");
654                        continue;
655                    }
656
657                    let (known, changeset) = {
658                        match process_single_version(&ctx.clock, ctx.metrics, &mut tx, change) {
659                            Ok(res) => res,
660                            Err(error) => {
661                                tracing::error!(%actor_id, versions = ?versions, %error, "error processing single version");
662
663                                if let Some(issue) = error
664                                    .sqlite_error_code()
665                                    .and_then(ChangeError::fatal_db_issue_for_code)
666                                {
667                                    super::fatal_db_issue(issue, &ctx.shutdown);
668                                }
669
670                                // the transaction was rolled back, so we need to return.
671                                if tx.is_autocommit() {
672                                    return Err(ChangeError::Rusqlite {
673                                        source: error,
674                                        actor_id: None,
675                                        version: None,
676                                    });
677                                } else {
678                                    continue;
679                                }
680                            }
681                        }
682                    };
683
684                    if let KnownDbVersion::Current(CurrentVersion { db_version, .. }) = &known {
685                        changesets.push((actor_id, changeset, *db_version, src));
686                    }
687
688                    known
689                };
690
691                let partial = match known {
692                    KnownDbVersion::Partial(partial) => Some(partial),
693                    _ => None,
694                };
695
696                seen.insert(versions.into(), partial.clone());
697                processed
698                    .entry(actor_id)
699                    .or_default()
700                    .push((versions, partial));
701            }
702        }
703
704        let elapsed = sub_start.elapsed();
705        if elapsed >= PROCESSING_WARN_THRESHOLD {
706            tracing::warn!(
707                ?elapsed,
708                "process_multiple_changes:: process_single_version took too long"
709            );
710        }
711
712        let sub_start = Instant::now();
713
714        let mut all_changes = Vec::with_capacity(processed.len());
715        let mut completed_versions = BTreeMap::new();
716        for (actor_id, processed) in processed.iter() {
717            let booked_write = booked_writes
718                .get_mut(actor_id)
719                .expect("booked write guard should be present for every actor");
720
721            let (changes, completed_partials) = booked_write.compute_and_apply(processed);
722            let clear_versions = changes
723                .clear_versions
724                .iter()
725                .map(|v| *v..=*v)
726                .collect::<RangeInclusiveSet<CrsqlDbVersion>>();
727            completed_versions.insert(*actor_id, (completed_partials, clear_versions));
728            all_changes.push(changes);
729        }
730
731        corro_types::bookie::BookieDbParams::from_changes(&all_changes)
732            .execute(&tx)
733            .map_err(|source| ChangeError::Rusqlite {
734                source,
735                actor_id: None,
736                version: None,
737            })?;
738
739        tx.commit().map_err(|source| {
740            if let Some(issue) = source
741                .sqlite_error_code()
742                .and_then(ChangeError::fatal_db_issue_for_code)
743            {
744                super::fatal_db_issue(issue, &ctx.shutdown);
745            }
746
747            ChangeError::Rusqlite {
748                source,
749                actor_id: None,
750                version: None,
751            }
752        })?;
753
754        let elapsed = sub_start.elapsed();
755        if elapsed >= PROCESSING_WARN_THRESHOLD {
756            tracing::warn!(
757                ?elapsed,
758                "process_multiple_changes: commiting transaction took too long"
759            );
760        }
761
762        for (_, booked_write) in booked_writes {
763            booked_write.commit();
764        }
765
766        // for (_, changeset, _, _) in changesets.iter() {
767        //     if let Some(ts) = changeset.ts() {
768        //         let dur = (agent.clock().new_timestamp().get_time() - ts.0).to_duration();
769        //         histogram!("corro.agent.changes.commit.lag.seconds").record(dur);
770        //         agent.metrics_tracker().observe_lag(dur.as_secs_f64());
771        //     }
772        // }
773
774        // schedule apply for partials that became complete
775        for (actor_id, (versions, clear_versions)) in completed_versions {
776            // try to send synchronously since we're in a blocking task and spawning a task to send a message is kind of gross
777            match ctx.apply_tx.try_reserve() {
778                Ok(permit) => permit.send((actor_id, versions)),
779                Err(mpsc::error::TrySendError::Full(_)) => {
780                    let tx = ctx.apply_tx.clone();
781                    tokio::task::spawn(async move {
782                        if tx.send((actor_id, versions)).await.is_err() {
783                            tracing::error!(
784                                "unable to apply further changes, receiver was closed after task spawn"
785                            );
786                        }
787                    });
788                }
789                Err(_) => {
790                    tracing::error!("unable to apply further changes, receiver was closed");
791                }
792            }
793
794            // the corrosion code seems to think this one isn't critical, or that it won't usually ever be full?
795            if ctx
796                .clear_buf_tx
797                .try_send((actor_id, clear_versions))
798                .is_err()
799            {
800                tracing::warn!("could not schedule buffered meta clear");
801            }
802        }
803
804        Ok::<_, ChangeError>(changesets)
805    })?;
806
807    let change_chunk_size: usize = changesets.iter().map(|cs| cs.1.len()).sum();
808
809    ctx.metrics
810        .changes_processed(start.elapsed(), "remote", Some(change_chunk_size));
811
812    tokio::spawn(async move {
813        for (_actor_id, changeset, db_version, _src) in changesets {
814            corro_types::updates::match_changes(&ctx.sub_manager, &changeset, db_version);
815            corro_types::updates::match_changes(&ctx.update_manager, &changeset, db_version);
816        }
817    });
818
819    Ok(())
820}

822fn process_single_version(
823    clock: &Clock,
824    metrics: &'static GossipMetrics,
825    tx: &mut sqlite_pool::InterruptibleTransaction<rusqlite::Transaction<'_>>,
826    change: ChangeV1,
827) -> rusqlite::Result<(KnownDbVersion, bx::Changeset)> {
828    let ChangeV1 {
829        actor_id,
830        changeset,
831    } = change;
832
833    let versions = changeset.versions();
834
835    let sp = tx.savepoint()?;
836    let mut changes_per_table = BTreeMap::new();
837    let (known, changeset) = if changeset.is_complete() {
838        let (known, changeset) = process_complete_version(
839            clock,
840            &sp,
841            actor_id,
842            versions,
843            changeset
844                .into_parts()
845                .expect("no changeset parts, this shouldn't be happening!"),
846            &mut changes_per_table,
847        )?;
848
849        (known, changeset)
850    } else {
851        let parts = changeset.into_parts().unwrap();
852        let known = process_incomplete_version(&sp, &parts, &mut changes_per_table)?;
853
854        (known, parts.into())
855    };
856
857    sp.commit()?;
858
859    for (table_name, count) in changes_per_table {
860        metrics.changes_committed_inc(&table_name, count, "remote");
861    }
862
863    Ok((known, changeset))
864}

866fn process_complete_version(
867    clock: &Clock,
868    sp: &sqlite_pool::InterruptibleTransaction<rusqlite::Savepoint<'_>>,
869    actor_id: ActorId,
870    versions: CrsqlDbVersionRange,
871    parts: ChangesetParts,
872    changes_per_table: &mut BTreeMap<String, u64>,
873) -> rusqlite::Result<(KnownDbVersion, Changeset)> {
874    let ChangesetParts {
875        version,
876        changes,
877        seqs,
878        last_seq,
879        ts,
880    } = parts;
881
882    let len = changes.len();
883
884    debug_assert!(
885        len <= seqs.len(),
886        "change from actor {actor_id} version {version} has len {len} but seqs range is {seqs:?} and last_seq is {last_seq}"
887    );
888
889    // Insert all the changes in a single statement
890    // This will return a non zero rowid only if the change impacted the database
891    let mut stmt = sp.prepare_cached(
892        r#"
893    INSERT INTO crsql_changes ("table", pk, cid, val, col_version, db_version, site_id, cl, seq, ts)
894        SELECT value0, value1, value2, value3, value4, value5, value6, value7, value8, value9
895        FROM unnest(?, ?, ?, ?, ?, ?, ?, ?, ?, ?)
896        -- WARNING: This returns a row BEFORE inserting not after
897        RETURNING db_version, seq, last_insert_rowid()
898    "#,
899    )?;
900    let params = rusqlite::params![
901        unnest_param(changes.iter().map(|c| c.table.as_str())),
902        unnest_param(changes.iter().map(|c| &c.pk)),
903        unnest_param(changes.iter().map(|c| c.cid.as_str())),
904        unnest_param(changes.iter().map(|c| &c.val)),
905        unnest_param(changes.iter().map(|c| &c.col_version)),
906        unnest_param(changes.iter().map(|c| &c.db_version)),
907        unnest_param(changes.iter().map(|c| &c.site_id)),
908        unnest_param(changes.iter().map(|c| &c.cl)),
909        unnest_param(changes.iter().map(|c| &c.seq)),
910        unnest_param(changes.iter().map(|_| &ts)),
911    ];
912    let mut last_rowids = stmt
913        .query_map(params, |row| Ok((row.get(0)?, row.get(1)?, row.get(2)?)))?
914        .collect::<rusqlite::Result<Vec<(CrsqlDbVersion, CrsqlSeq, i64)>>>()?;
915
916    let last_rowids_len = last_rowids.len();
917
918    // RETURNING returns rows BEFORE the insert, not after
919    // so the rowids we get will be shifted by one
920    // we need to shift them back
921    if last_rowids_len > 0 {
922        for i in 0..(last_rowids_len - 1) {
923            last_rowids[i].2 = last_rowids[i + 1].2;
924        }
925        last_rowids[last_rowids_len - 1].2 = sp
926            .prepare_cached("SELECT last_insert_rowid()")?
927            .query_row(rusqlite::params![], |row| row.get(0))?;
928    }
929
930    // Now determine which exact changes impacted the database
931    // This is mostly for keeping accurate metrics
932    let mut impactful_changeset = vec![];
933    for (change, (db_version, seq, rowid)) in changes.into_iter().zip(last_rowids) {
934        // Those asserts are only a sanity check
935        assert!(db_version == change.db_version);
936        assert!(seq == change.seq);
937
938        if rowid == 0 {
939            continue;
940        }
941
942        let table_name = change.table.0.to_string();
943        impactful_changeset.push(change);
944        *changes_per_table.entry(table_name).or_default() += 1;
945    }
946
947    let (known_version, new_changeset) = if impactful_changeset.is_empty() {
948        (
949            KnownDbVersion::Cleared,
950            Changeset::Empty {
951                versions,
952                ts: Some(clock.new_timestamp()),
953            },
954        )
955    } else {
956        (
957            KnownDbVersion::Current(CurrentVersion {
958                db_version: version,
959                last_seq,
960                ts,
961            }),
962            Changeset::Full {
963                version,
964                changes: impactful_changeset,
965                seqs,
966                last_seq,
967                ts,
968            },
969        )
970    };
971
972    Ok((known_version, new_changeset))
973}

975fn process_incomplete_version(
976    sp: &sqlite_pool::InterruptibleTransaction<rusqlite::Savepoint<'_>>,
977    parts: &ChangesetParts,
978    changes_per_table: &mut BTreeMap<String, u64>,
979) -> rusqlite::Result<KnownDbVersion> {
980    let ChangesetParts {
981        changes,
982        seqs,
983        last_seq,
984        ts,
985        ..
986    } = parts;
987
988    let mut stmt = sp.prepare_cached(
989                r#"
990                INSERT INTO __corro_buffered_changes
991                    ("table", pk, cid, val, col_version, db_version, site_id, cl, seq, ts)
992                SELECT
993                    value0, value1, value2, value3, value4, value5, value6, value7, value8, value9
994                FROM
995                    unnest(:table_arr, :pk_arr, :cid_arr, :val_arr, :col_version_arr, :db_version_arr, :site_id_arr, :cl_arr, :seq_arr, :ts_arr)
996                -- Otherwise sqlite will think ON CONFLICT is part of a JOIN
997                WHERE TRUE
998                ON CONFLICT (site_id, db_version, seq)
999                    DO NOTHING
1000                RETURNING
1001                    "table"
1002            "#,
1003            )?;
1004    let table_names = stmt.query_map(
1005        rusqlite::named_params! {
1006            ":table_arr": unnest_param(changes.iter().map(|change| change.table.as_str())),
1007            ":pk_arr": unnest_param(changes.iter().map(|change| &change.pk)),
1008            ":cid_arr": unnest_param(changes.iter().map(|change| change.cid.as_str())),
1009            ":val_arr": unnest_param(changes.iter().map(|change| &change.val)),
1010            ":col_version_arr": unnest_param(changes.iter().map(|change| change.col_version)),
1011            ":db_version_arr": unnest_param(changes.iter().map(|change| change.db_version)),
1012            ":site_id_arr": unnest_param(changes.iter().map(|change| &change.site_id)),
1013            ":cl_arr": unnest_param(changes.iter().map(|change| change.cl)),
1014            ":seq_arr": unnest_param(changes.iter().map(|change| change.seq)),
1015            ":ts_arr": unnest_param(changes.iter().map(|_| ts)),
1016        },
1017        |row| row.get::<_, String>(0),
1018    )?;
1019
1020    for res in table_names {
1021        let tn = res?;
1022        *changes_per_table.entry(tn).or_default() += 1;
1023    }
1024
1025    Ok(KnownDbVersion::Partial(PartialVersion {
1026        seqs: RangeInclusiveSet::from_iter([(*seqs).into()]),
1027        last_seq: *last_seq,
1028        ts: *ts,
1029    }))
1030}

1034fn match_changes_from_db_version<H>(
1035    manager: &impl corro_types::updates::Manager<H>,
1036    conn: &rusqlite::Connection,
1037    db_version: CrsqlDbVersion,
1038    actor_id: ActorId,
1039) -> rusqlite::Result<()>
1040where
1041    H: corro_types::updates::Handle + Send + 'static,
1042{
1043    let handles = manager.get_handles();
1044    if handles.is_empty() {
1045        return Ok(());
1046    }
1047
1048    let mut candidates = handles
1049        .iter()
1050        .map(|(id, handle)| (id, (corro_types::pubsub::MatchCandidates::new(), handle)))
1051        .collect::<BTreeMap<_, _>>();
1052
1053    {
1054        let mut prepped = conn.prepare_cached(
1055            r#"
1056        SELECT "table", pk, cid, cl
1057            FROM crsql_changes
1058            WHERE db_version = ?
1059              AND site_id = ?
1060            ORDER BY seq ASC
1061        "#,
1062        )?;
1063
1064        let rows = prepped.query_map((db_version, actor_id), |row| {
1065            Ok((
1066                row.get::<_, corro_api_types::TableName>(0)?,
1067                row.get::<_, Vec<u8>>(1)?,
1068                row.get::<_, corro_api_types::ColumnName>(2)?,
1069                row.get::<_, i64>(3)?,
1070            ))
1071        })?;
1072
1073        for change_res in rows {
1074            let (table, pk, column, cl) = change_res?;
1075
1076            for (candidates, handle) in candidates.values_mut() {
1077                let change = corro_types::pubsub::MatchableChange {
1078                    table: &table,
1079                    pk: &pk,
1080                    column: &column,
1081                    cl,
1082                };
1083                handle.filter_matchable_change(candidates, change);
1084            }
1085        }
1086    }
1087
1088    // The original corrosion code spawns individual tasks for each case of the channel being full or closed, we just spawn
1089    // a single task for each category
1090    let mut full = Vec::new();
1091    let mut closed = Vec::new();
1092
1093    for (id, (candidates, handle)) in candidates {
1094        let mut match_count = 0;
1095        for (table, pks) in candidates.iter() {
1096            let count = pks.len();
1097            match_count += count;
1098            handle
1099                .get_counter(table)
1100                .matched_count
1101                .increment(pks.len() as u64);
1102        }
1103
1104        tracing::trace!(sub_id = %id, %db_version, match_count, "found candidates");
1105
1106        let Err(err) = handle.changes_tx().try_send(candidates) else {
1107            continue;
1108        };
1109
1110        match err {
1111            mpsc::error::TrySendError::Full(item) => {
1112                full.push((handle.changes_tx(), item));
1113            }
1114            mpsc::error::TrySendError::Closed(_) => {
1115                let Some(handle) = manager.remove(id) else {
1116                    continue;
1117                };
1118
1119                closed.push(handle);
1120            }
1121        }
1122    }
1123
1124    if !full.is_empty() {
1125        tokio::spawn(async move {
1126            let total = full.len();
1127
1128            let start = Instant::now();
1129
1130            let mut sends = futures::stream::iter(
1131                full.into_iter()
1132                    .map(|(tx, item)| async move { tx.send(item).await }),
1133            )
1134            .buffer_unordered(10);
1135
1136            let mut errors = 0;
1137
1138            while let Some(res) = sends.next().await {
1139                if res.is_err() {
1140                    errors += 1;
1141                }
1142            }
1143
1144            tracing::debug!(total, errors, elapsed = ?start.elapsed(), "asynchronously sent changes to channels which were full");
1145        });
1146    }
1147
1148    if !closed.is_empty() {
1149        tokio::spawn(async move {
1150            for handle in closed {
1151                handle.cleanup().await;
1152            }
1153        });
1154    }
1155
1156    Ok(())
1157}

1159async fn process_fully_buffered_changes(
1160    ctx: &ChangeCtx,
1161    bookie: &Bookie,
1162    actor_id: ActorId,
1163    version: CrsqlDbVersion,
1164    tx_timeout: Duration,
1165) -> Result<bool, ChangeError> {
1166    let rows_impacted = {
1167        let booked = bookie.ensure(actor_id);
1168        let mut conn = ctx.pool.write_normal().await?;
1169        tracing::trace!(%actor_id, %version, "acquired write (normal) connection to process fully buffered changes");
1170
1171        tokio::task::block_in_place(move || {
1172            let bookie_write = bookie.write_lock_blocking();
1173            let bookedw = bookie_write.write_tx(&booked);
1174            tracing::trace!(%actor_id, %version, "acquired Booked write lock to process fully buffered changes");
1175
1176            let mut bookedw = bookedw;
1177            let last_seq = if let Some(PartialVersion { seqs, last_seq, .. }) =
1178                bookedw.partials.get(&version)
1179            {
1180                if seqs.gaps(&(CrsqlSeq(0)..=*last_seq)).count() != 0 {
1181                    let gaps = seqs
1182                        .gaps(&(CrsqlSeq(0)..=*last_seq))
1183                        .collect::<RangeInclusiveSet<CrsqlSeq>>();
1184                    tracing::error!(%actor_id, %version, ?gaps, "found sequence gaps, aborting!", );
1185                    // TODO: return an error here
1186                    return Ok(false);
1187                }
1188                *last_seq
1189            } else {
1190                tracing::warn!(%actor_id, %version, "version not found in cache, returning");
1191                return Ok(false);
1192            };
1193
1194            let base_tx = conn
1195                .immediate_transaction()
1196                .map_err(|source| ChangeError::Rusqlite {
1197                    source,
1198                    actor_id: Some(actor_id),
1199                    version: Some(version),
1200                })?;
1201
1202            let tx = sqlite_pool::InterruptibleTransaction::new(
1203                base_tx,
1204                Some(tx_timeout),
1205                "process_buffered_changes",
1206            );
1207
1208            tracing::trace!(%actor_id, %version, %last_seq, "Processing buffered changes to crsql_changes");
1209
1210            let rows_present: bool = tx.prepare_cached("SELECT EXISTS (SELECT 1 FROM __corro_buffered_changes WHERE site_id = ? AND db_version = ?)")
1211                                    .map_err(|source| ChangeError::Rusqlite{source, actor_id: Some(actor_id), version: Some(version)})?
1212                                    .query_row(rusqlite::params![actor_id, version], |row| row.get(0))
1213                                    .map_err(|source| ChangeError::Rusqlite{source, actor_id: Some(actor_id), version: Some(version)})?;
1214
1215            let start = Instant::now();
1216
1217            if rows_present {
1218                // insert all buffered changes into crsql_changes directly from the buffered changes table
1219                let count = tx
1220                    .prepare_cached(
1221                        r#"
1222                        INSERT INTO crsql_changes ("table", pk, cid, val, col_version, db_version, site_id, cl, seq, ts)
1223                            SELECT                 "table", pk, cid, val, col_version, db_version, site_id, cl, seq, ts
1224                                FROM __corro_buffered_changes
1225                                    WHERE site_id = ?
1226                                    AND db_version = ?
1227                                    ORDER BY db_version ASC, seq ASC
1228                                    "#,
1229                    ).map_err(|source| ChangeError::Rusqlite{source, actor_id: Some(actor_id), version: Some(version)})?
1230                    .execute(rusqlite::params![actor_id.as_bytes(), version]).map_err(|source| ChangeError::Rusqlite{source, actor_id: Some(actor_id), version: Some(version)})?;
1231                tracing::trace!(%actor_id, %version, count, elapsed = ?start.elapsed(), "Inserted rows from buffered into crsql_changes");
1232            } else {
1233                tracing::trace!(%actor_id, %version, "No buffered rows, skipped insertion into crsql_changes");
1234            }
1235
1236            let rows_impacted: i64 = tx
1237                .prepare_cached("SELECT crsql_rows_impacted()")
1238                .map_err(|source| ChangeError::Rusqlite {
1239                    source,
1240                    actor_id: Some(actor_id),
1241                    version: Some(version),
1242                })?
1243                .query_row((), |row| row.get(0))
1244                .map_err(|source| ChangeError::Rusqlite {
1245                    source,
1246                    actor_id: Some(actor_id),
1247                    version: Some(version),
1248                })?;
1249
1250            tracing::trace!(%actor_id, %version, rows_impacted, "rows impacted by buffered changes insertion");
1251
1252            bookedw
1253                .insert_db(&tx, [version..=version].into())
1254                .map_err(|source| ChangeError::Rusqlite {
1255                    source,
1256                    actor_id: Some(actor_id),
1257                    version: Some(version),
1258                })?;
1259
1260            // the version transitions from partial → fully applied, clean up its seq bookkeeping
1261            bookedw
1262                .clear_partials(&tx, RangeInclusiveSet::from([version..=version]))
1263                .map_err(|source| ChangeError::Rusqlite {
1264                    source,
1265                    actor_id: Some(actor_id),
1266                    version: Some(version),
1267                })?;
1268
1269            tx.commit().map_err(|source| ChangeError::Rusqlite {
1270                source,
1271                actor_id: Some(actor_id),
1272                version: Some(version),
1273            })?;
1274
1275            bookedw.commit();
1276
1277            if let Err(error) = ctx
1278                .clear_buf_tx
1279                .try_send((actor_id, rangemap::range_inclusive_set![version..=version]))
1280            {
1281                tracing::error!(%error, "could not schedule buffered data clear");
1282            }
1283
1284            Ok::<_, ChangeError>(rows_impacted > 0)
1285        })
1286    }?;
1287
1288    if rows_impacted {
1289        let conn = ctx.pool.read().await?;
1290        tokio::task::block_in_place(|| {
1291            if let Err(error) =
1292                match_changes_from_db_version(&ctx.sub_manager, &conn, version, actor_id)
1293            {
1294                tracing::error!(%error, %version, "could not match changes for subs from db version");
1295            }
1296        });
1297
1298        tokio::task::block_in_place(|| {
1299            if let Err(error) =
1300                match_changes_from_db_version(&ctx.update_manager, &conn, version, actor_id)
1301            {
1302                tracing::error!(%error, %version, "could not match changes for updates from db version");
1303            }
1304        });
1305    }
1306
1307    Ok(rows_impacted)
1308}

1310pub fn spawn_buffered_apply(
1311    ctx: ChangeCtx,
1312    opts: ChangeOptions,
1313    bookie: Bookie,
1314    mut apply_rx: mpsc::Receiver<ChangeApply>,
1315    mut tripwire: tripwire::Tripwire,
1316) {
1317    async fn find_fully_buffered_partial(
1318        pool: &corro_types::agent::SplitPool,
1319    ) -> Result<Option<(ActorId, CrsqlDbVersion)>, ChangeError> {
1320        let conn = pool.read().await.map_err(ChangeError::SqlitePool)?;
1321
1322        tokio::task::block_in_place(|| {
1323            conn.prepare_cached(
1324                "SELECT site_id, db_version FROM __corro_seq_bookkeeping
1325                 WHERE start_seq = 0 AND end_seq = last_seq
1326                 LIMIT 1",
1327            )
1328            .and_then(
1329                |mut stmt| match stmt.query_row([], |row| Ok((row.get(0)?, row.get(1)?))) {
1330                    Ok(v) => Ok(Some(v)),
1331                    Err(rusqlite::Error::QueryReturnedNoRows) => Ok(None),
1332                    Err(e) => Err(e),
1333                },
1334            )
1335            .map_err(|source| ChangeError::Rusqlite {
1336                source,
1337                actor_id: None,
1338                version: None,
1339            })
1340        })
1341    }
1342
1343    spawn::spawn_counted(async move {
1344        // we can burst timeout up to an additional 2 min
1345        let max_timeout_increase: u64 = 6;
1346        let step_timeout_secs: u64 = 20;
1347
1348        let throttle_min = Duration::from_secs(5 * 60);
1349        let throttle_max = Duration::from_secs(60 * 60);
1350
1351        let mut retry_interval = tokio::time::interval(Duration::from_secs(5 * 60));
1352
1353        // map to throttle retries for failed versions that took too long to apply
1354        let mut limit_retries = throttle::ThrottleMap::new(throttle_min, throttle_max);
1355
1356        let do_process =
1357            async |partial_version: (ActorId, CrsqlDbVersion),
1358                   limit_retries: &mut throttle::ThrottleMap| {
1359                async {
1360                if let Some(blocked_until) = limit_retries.is_throttled(&partial_version) {
1361                    let next_retry = blocked_until.duration_since(Instant::now());
1362                    tracing::debug!(
1363                        ?next_retry,
1364                        "previous attempt to apply buffered changes took too long, will retry"
1365                    );
1366                    return;
1367                }
1368
1369                let throttle_count = limit_retries
1370                    .throttle_count(&partial_version)
1371                    .min(max_timeout_increase);
1372                let tx_timeout =
1373                    opts.sql_tx_timeout + Duration::from_secs(step_timeout_secs * throttle_count);
1374
1375                tracing::debug!(?tx_timeout, "picked up buffered changes");
1376
1377                let start = Instant::now();
1378                let res =
1379                    process_fully_buffered_changes(&ctx, &bookie, partial_version.0, partial_version.1, tx_timeout)
1380                        .await;
1381                let elapsed = start.elapsed();
1382                match res {
1383                    Ok(false) => {
1384                        tracing::debug!("did not apply buffered changes");
1385                        limit_retries.remove(&partial_version);
1386                    }
1387                    Ok(true) => {
1388                        tracing::debug!("succesfully applied buffered changes");
1389                        ctx.metrics.changes_processed(elapsed, "buffered", None);
1390                        limit_retries.remove(&partial_version);
1391                    }
1392                    Err(error) => {
1393                        let is_interrupt_error = error.is_interrupt_error();
1394                        tracing::debug!(elapsed = ?tx_timeout, %error, "could not apply fully buffered changes");
1395                        if let Some(issue) = error.fatal_db_issue() {
1396                            super::fatal_db_issue(issue, &ctx.shutdown);
1397                        }
1398
1399                        // processing time came close to timeout, limit retry with exponential backoff
1400                        if is_interrupt_error {
1401                            limit_retries.throttle(partial_version);
1402                        }
1403                    }
1404                }
1405            }.instrument(tracing::debug_span!("background apply", actor_id = %partial_version.0, version = %partial_version.1)).await;
1406            };
1407
1408        retry_interval.tick().await;
1409
1410        loop {
1411            // TODO: The original corrosion code was sending one change at a time but I changed it so it was n changes per
1412            // actor instead which would be beneficial in this write code by consolidating DB access
1413            tokio::select! {
1414                biased;
1415                _ = &mut tripwire => break,
1416                maybe_version = apply_rx.recv() => {
1417                    let Some((actor, versions)) = maybe_version else {
1418                        break;
1419                    };
1420
1421                    for version in versions {
1422                        do_process((actor, version), &mut limit_retries).await;
1423                    }
1424                },
1425                _ = retry_interval.tick() => {
1426                    match find_fully_buffered_partial(&ctx.pool).await {
1427                        Ok(Some(pair)) => {
1428                            do_process(pair, &mut limit_retries).await;
1429                        }
1430                        Ok(None) => {}
1431                        Err(error) => {
1432                            tracing::debug!(%error, "could not query for fully buffered partials");
1433                        },
1434                    }
1435                },
1436            }
1437        }
1438    });
1439}

1441/// Clean up `__corro_buffered_changes` rows for versions that have been fully applied.
1442/// `__corro_seq_bookkeeping` is managed through the bookie via `insert_partials_db`/`insert_db`.
1443pub fn spawn_buffered_cleanup(
1444    ctx: ChangeCtx,
1445    opts: ChangeOptions,
1446    mut clear_buf_rx: mpsc::Receiver<(ActorId, RangeInclusiveSet<CrsqlDbVersion>)>,
1447    mut tripwire: tripwire::Tripwire,
1448) {
1449    async fn find_orphaned_buffered_changes(
1450        pool: &corro_types::agent::SplitPool,
1451    ) -> Result<Option<(ActorId, CrsqlDbVersion)>, ChangeError> {
1452        let conn = pool.read().await.map_err(ChangeError::SqlitePool)?;
1453        tokio::task::block_in_place(|| {
1454            conn.prepare_cached(
1455                "SELECT DISTINCT bc.site_id, bc.db_version
1456                 FROM __corro_buffered_changes bc
1457                 WHERE NOT EXISTS (
1458                     SELECT 1 FROM __corro_seq_bookkeeping bk
1459                     WHERE bk.site_id = bc.site_id AND bk.db_version = bc.db_version
1460                 )
1461                 LIMIT 1",
1462            )
1463            .and_then(
1464                |mut stmt| match stmt.query_row([], |row| Ok((row.get(0)?, row.get(1)?))) {
1465                    Ok(v) => Ok(Some(v)),
1466                    Err(rusqlite::Error::QueryReturnedNoRows) => Ok(None),
1467                    Err(e) => Err(e),
1468                },
1469            )
1470            .map_err(|source| ChangeError::Rusqlite {
1471                source,
1472                actor_id: None,
1473                version: None,
1474            })
1475        })
1476    }
1477
1478    spawn::spawn_counted(async move {
1479        // check for orphaned buffered changes every 5 minutes
1480        let mut retry_interval = tokio::time::interval(Duration::from_secs(5 * 60));
1481        retry_interval.tick().await;
1482
1483        loop {
1484            let (actor_id, versions) = tokio::select! {
1485                biased;
1486                _ = &mut tripwire => break,
1487                maybe_partials = clear_buf_rx.recv() => {
1488                    let Some(partials) = maybe_partials else {
1489                        break;
1490                    };
1491
1492                    partials
1493                }
1494                _ = retry_interval.tick() => {
1495                    match find_orphaned_buffered_changes(&ctx.pool).await {
1496                        Ok(Some((actor_id, version))) => (actor_id, rangemap::range_inclusive_set![version..=version]),
1497                        Ok(None) => continue,
1498                        Err(error) => {
1499                            tracing::debug!(%error, "could not query for orphaned buffered changes");
1500                            continue;
1501                        }
1502                    }
1503                }
1504            };
1505
1506            let to_clear = if let Some(booked) = ctx.bookie.get(&actor_id) {
1507                let booked_read = booked.read();
1508                // The original corrosion code seems to have a bug where a continue is done on the inner loop instead of the
1509                // outer loop
1510                // for version in versions {
1511                //     if booked_read.get_partial(&version).is_some() {
1512                //         warn!(%actor_id, %version, "clear_buffered_meta: received partial version that's still in bookie, skipping");
1513                //         continue;
1514                //     }
1515                // }
1516
1517                let before = versions.len();
1518
1519                // This is a poor version of retain since rangemap does not have that functionality
1520                let to_clear: rangemap::RangeInclusiveSet<_> = versions
1521                    .into_iter()
1522                    .filter(|v| {
1523                        for v in v.start().0..=v.end().0 {
1524                            if booked_read.get_partial(&CrsqlDbVersion(v)).is_some() {
1525                                return false;
1526                            }
1527                        }
1528
1529                        true
1530                    })
1531                    .collect();
1532
1533                if to_clear.is_empty() {
1534                    tracing::debug!(%actor_id, "received set of partial versions that were all still in the bookie, skipping");
1535                    continue;
1536                } else if before > to_clear.len() {
1537                    tracing::debug!(%actor_id, still_booked = before - to_clear.len(), "received set of partial versions, some of which were still booked");
1538                }
1539
1540                to_clear
1541            } else {
1542                versions
1543            };
1544
1545            let pool = ctx.pool.clone();
1546            let tx_timeout = opts.sql_tx_timeout;
1547            let self_actor_id = ctx.actor_id;
1548            let mut task_tripwire = tripwire.clone();
1549
1550            const TO_CLEAR_COUNT: usize = 1000;
1551
1552            spawn::spawn_counted(async move {
1553                loop {
1554                    let res = {
1555                        let mut conn = pool.write_low().await?;
1556
1557                        tokio::task::block_in_place(|| {
1558                            let tx = sqlite_pool::InterruptibleTransaction::new(
1559                                conn.immediate_transaction()?,
1560                                Some(tx_timeout),
1561                                "clear_buffered_meta",
1562                            );
1563
1564                            let mut buf_count = 0;
1565
1566                            for range in to_clear.iter() {
1567                                buf_count += tx
1568                                .prepare_cached("DELETE FROM __corro_buffered_changes WHERE (site_id, db_version, seq) IN (SELECT site_id, db_version, seq FROM __corro_buffered_changes WHERE site_id = ? AND db_version >= ? AND db_version <= ? LIMIT ?)")?
1569                                .execute(rusqlite::params![actor_id, range.start().0, range.end().0, TO_CLEAR_COUNT])?;
1570                            }
1571
1572                            tx.commit()?;
1573
1574                            Ok::<_, rusqlite::Error>(buf_count)
1575                        })
1576                    };
1577
1578                    match res {
1579                        Ok(buf_count) => {
1580                            if buf_count > 0 {
1581                                tracing::debug!(buf_count, "cleared buffered change rows");
1582                            }
1583                            if buf_count < TO_CLEAR_COUNT {
1584                                break;
1585                            }
1586                        }
1587                        Err(error) => {
1588                            tracing::debug!(%error, "could not clear buffered meta for versions");
1589                        }
1590                    }
1591
1592                    tokio::select! {
1593                        _ = &mut task_tripwire => {
1594                            break;
1595                        }
1596                        _ = tokio::time::sleep(Duration::from_secs(2)) => {}
1597                    }
1598                }
1599
1600                Ok::<_, eyre::Report>(())
1601            }.instrument(tracing::debug_span!("clear buffered version", %actor_id, %self_actor_id)));
1602        }
1603    });
