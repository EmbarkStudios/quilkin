12/// Wraps [`SplitPool::read`] to enforce `temp_store = MEMORY` and `query_only = ON`.
13/// Use this instead of [`SplitPool::read`] directly.

15    fn read_readonly(
16        &self,
17    ) -> impl std::future::Future<Output = Result<sqlite_pool::Connection<CrConn>, SqlitePoolError>> + Send;
18}

20impl SplitPoolReadExt for SplitPool {
21    async fn read_readonly(&self) -> Result<sqlite_pool::Connection<CrConn>, SqlitePoolError> {
22        let conn = self.read().await?;
23        // flag-only pragmas; failure means the connection itself is broken
24        conn.pragma_update(None, "temp_store", "MEMORY")
25            .expect("temp_store is a flag pragma; failure means the connection is broken");
26        conn.pragma_update(None, "query_only", true)
27            .expect("query_only is a flag pragma; failure means the connection is broken");
28        Ok(conn)
29    }
30}

32/// WAL synchronous mode. Affects whether `fsync` is called after each commit.
33///
34/// In WAL mode [`Normal`][WalSynchronous::Normal] is safe from database corruption (data can only
35/// be lost on a simultaneous OS crash + hardware failure) and avoids the per-commit `fsync`.
36/// For corrosion nodes that replicate state across peers, [`Normal`][WalSynchronous::Normal] is
37/// the recommended setting.
38#[derive(Clone, Copy, Default, Debug)]
39pub enum WalSynchronous {
40    /// Default `SQLite` WAL behaviour: `fsync` after every commit. No data loss on power failure.
41    #[default]
42    Full,
43    /// Skip per-commit `fsync`; only sync during WAL checkpoints. Eliminates commit stalls on
44    /// slow fsync paths (e.g. WSL2, network filesystems). Safe when data is replicated.
45    Normal,
46}
47
48impl WalSynchronous {
49    fn as_str(self) -> &'static str {
50        match self {
51            Self::Full => "FULL",
52            Self::Normal => "NORMAL",
53        }
54    }
55}

57/// Hard size limits. Volume allocation must exceed `(max_page_count × page_size) + journal_size_limit`.
58#[derive(Default)]
59pub struct DBLimits {
60    /// Max database file size in pages (4 KiB each; 262,144 = 1 GiB). `None` leaves the built-in ceiling.
61    pub max_page_count: Option<u64>,
62    /// Max WAL size in bytes after a checkpoint. `None` leaves the `SQLite` default (unlimited).
63    pub journal_size_limit: Option<i64>,
64    /// WAL synchronous mode. `None` leaves the startup setting ([`WalSynchronous::Normal`]).
65    pub synchronous: Option<WalSynchronous>,
66    /// Auto-checkpoint threshold in WAL pages. `Some(0)` disables automatic checkpointing entirely,
67    /// deferring all WAL cleanup to the manual [`wal_checkpoint_over_threshold`] path.
68    /// `None` leaves the `SQLite` default (1000 pages ≈ 4 MiB).
69    pub wal_autocheckpoint: Option<u32>,
70}
71
72/// Percentage of the volume the main database file may occupy.
73const DB_CAPACITY_PCT: u64 = 75;
74/// Percentage of the volume the WAL may retain after a checkpoint.
75const WAL_CAPACITY_PCT: u64 = 12;
76
77/// Caps for a volume of `capacity` bytes, as `(max_page_count, journal_size_limit)`.
78///
79/// The two together are held under 100% so the remainder stays free for recovery writes.
80/// `None` if the volume cannot fit even a single page under the database cap.
81fn derive_limits(capacity: u64, page_size: u64) -> Option<(u64, i64)> {
82    let page_size = page_size.max(1);
83    let max_page_count = capacity.checked_mul(DB_CAPACITY_PCT)? / 100 / page_size;
84    let journal_size_limit = capacity.checked_mul(WAL_CAPACITY_PCT)? / 100;
85    (max_page_count > 0).then_some((max_page_count, journal_size_limit as i64))
86}

88/// Total size in bytes of the filesystem holding `path`.
89#[cfg(unix)]
90fn volume_capacity(path: &crate::Path) -> Option<u64> {
91    let stat = rustix::fs::statvfs(path.as_str()).ok()?;
92    stat.f_blocks.checked_mul(stat.f_frsize)
93}
94
95#[cfg(not(unix))]
96fn volume_capacity(_path: &crate::Path) -> Option<u64> {
97    None
98}

100impl DBLimits {
101    /// Fills whichever size caps are unset from the capacity of the volume holding `db_path`.
102    ///
103    /// Left unset, `SQLite` grows until the filesystem is exhausted, and at that point
104    /// `wal_checkpoint`, `incremental_vacuum` and the row purge in [`recover_space`] all need
105    /// to write and so all fail: the database is wedged until an operator intervenes. Capping
106    /// the database and WAL below capacity turns that into `SQLITE_FULL` while free space
107    /// remains, which [`crate::persistent::mutator::Transactor::with_full_recovery`] can act on.
108    pub fn apply_volume_defaults(
109        &mut self,
110        conn: &rusqlite::Connection,
111        db_path: &crate::Path,
112    ) -> eyre::Result<()> {
113        if self.max_page_count.is_some() && self.journal_size_limit.is_some() {
114            return Ok(());
115        }
116
117        let Some(capacity) = volume_capacity(db_path) else {
118            tracing::warn!(%db_path, "volume capacity unavailable, database growth is unbounded");
119            return Ok(());
120        };
121
122        let page_size: i64 = conn.pragma_query_value(None, "page_size", |row| row.get(0))?;
123        let Some((max_page_count, journal_size_limit)) = derive_limits(capacity, page_size as u64)
124        else {
125            tracing::warn!(
126                capacity,
127                page_size,
128                "volume too small to derive size limits"
129            );
130            return Ok(());
131        };
132
133        self.max_page_count.get_or_insert(max_page_count);
134        self.journal_size_limit.get_or_insert(journal_size_limit);
135        tracing::info!(capacity, "derived size limits from volume capacity");
136
137        Ok(())
138    }
139}

155/// Checkpoints the WAL and runs incremental vacuum. If `purge_count` is `Some(n)`,
156/// deletes the `n` oldest `servers` rows first to free pages for reuse.
157pub async fn recover_space(pool: &SplitPool, purge_count: Option<usize>) -> eyre::Result<()> {
158    let conn = pool.write_priority().await?;
159    tokio::task::block_in_place(|| {
160        if let Some(count) = purge_count {
161            conn.execute(
162                "DELETE FROM servers WHERE endpoint IN \
163                 (SELECT endpoint FROM servers ORDER BY cont_update ASC LIMIT ?)",
164                rusqlite::params![count],
165            )?;
166            tracing::info!(count, "purged oldest servers to recover disk space");
167        }
168        let busy: bool = conn.query_row("PRAGMA wal_checkpoint(TRUNCATE)", [], |row| row.get(0))?;
169        if busy {
170            tracing::warn!("WAL checkpoint during space recovery was busy");
171        }
172        conn.execute_batch("PRAGMA incremental_vacuum")?;
173        Ok::<_, eyre::Error>(())
174    })?;
175    Ok(())
176}

178pub struct DBMaintenance {
179    /// The interval at which to run maintenance
180    pub interval: Duration,
181    /// The Write-Ahead-Log threshold, in MiB
182    pub wal_threshold: u32,
183    /// The maximum number of unused free pages that can exist (a page is 4KiB)
184    pub max_free_pages: u64,
185    /// How long a deleted row's tombstone is kept before [`reap_tombstones`] collects it.
186    ///
187    /// `None` disables collection, which lets the clock tables grow without bound under
188    /// row churn. Must exceed the time a partitioned peer can take to converge, otherwise
189    /// that peer can resurrect a row whose tombstone has already been collected.
190    pub tombstone_retention: Option<Duration>,
191    /// Tombstones collected per table per tick, bounding how long the write lock is held.
192    pub tombstone_batch: usize,
193    pub limits: DBLimits,
194}
195
196impl Default for DBMaintenance {
197    fn default() -> Self {
198        Self {
199            interval: Duration::from_secs(5 * 60),
200            // 2 GiB
201            wal_threshold: 2 * 1024,
202            max_free_pages: 10000,
203            // orders of magnitude above gossip convergence, and short enough that
204            // churn cannot accumulate a meaningful amount of dead weight
205            tombstone_retention: Some(Duration::from_secs(60 * 60)),
206            tombstone_batch: 2000,
207            limits: DBLimits::default(),
208        }
209    }
210}

225    pub actor_id: ActorId,

229    /// the same cluster id

233impl InitializedDb {
234    /// Attempts to initialize a CRSQL database with the specified schema, creating
235    /// it if it doesn't exist
236    ///
237    /// If `maintenance` is specified, spawns a task to maintain the WAL and vacuum the DB
238    pub async fn setup(
239        db_path: &crate::Path,
240        schema: &str,
241        mut maintenance: Option<DBMaintenance>,
242    ) -> eyre::Result<Self> {
243        let partial_schema = corro_types::schema::parse_sql(schema)?;
244
245        let actor_id = {
246            // we need to set auto_vacuum before any tables are created
247            let db_conn = rusqlite::Connection::open(db_path)?;
248            db_conn.execute_batch("PRAGMA auto_vacuum = INCREMENTAL")?;
249
250            let conn = CrConn::init(db_conn)?;
251            conn.query_row("SELECT crsql_site_id();", [], |row| {
252                row.get::<_, ActorId>(0)
253            })?
254        };
255
256        /// We might want to make this configurable, but for now just hardcode to 1GiB, note that this is a negative value
257        /// to inidicate it is in KiB, if it was positive it would be the cache size in pages
258        const ONE_GIB: i64 = -(1024 * 1024);
259
260        let write_sema = Arc::new(tokio::sync::Semaphore::new(1));
261        let pool = SplitPool::create(&db_path, write_sema.clone(), ONE_GIB).await?;
262
263        let clock = crate::Clock(Arc::new(
264            uhlc::HLCBuilder::default()
265                .with_id(actor_id.try_into().unwrap())
266                .with_max_delta(std::time::Duration::from_millis(300))
267                .build(),
268        ));
269
270        let schema = {
271            let mut conn = pool.write_priority().await?;
272
273            let old_schema = {
274                corro_types::agent::migrate(clock.0.clone(), &mut conn)?;
275                let mut schema = corro_types::schema::init_schema(&conn)?;
276                schema.constrain()?;
277
278                schema
279            };
280
281            let schema = tokio::task::block_in_place(|| {
282                update_schema(&mut conn, old_schema, partial_schema)
283            })?;
284
285            if let Some(dbm) = maintenance.as_mut() {
286                tokio::task::block_in_place(|| dbm.limits.apply_volume_defaults(&conn, db_path))?;
287            }
288
289            let limits = maintenance.as_ref().map(|m| &m.limits);
290            tokio::task::block_in_place(|| run_startup_checks(&conn, limits))?;
291
292            schema
293        };
294
295        let cluster_id = {
296            let conn = pool.read().await?;
297
298            match conn.query_row(
299                "SELECT value FROM __corro_state WHERE key = 'cluster_id'",
300                [],
301                |row| row.get(0),
302            ) {
303                Ok(value) => value,
304                Err(rusqlite::Error::QueryReturnedNoRows) => Default::default(),
305                Err(e) => return Err(e.into()),
306            }
307        };
308
309        if let Some(dbm) = maintenance {
310            let tables = schema.tables.keys().cloned().collect();
311            spawn_db_maintenance(db_path, pool.clone(), dbm, clock.clone(), tables);
312        }
313
314        use eyre::WrapErr;
315        let start = std::time::Instant::now();
316        let conn = pool
317            .read_readonly()
318            .await
319            .context("failed to acquire read")?;
320
321        let (bookie, booked) =
322            tokio::task::block_in_place(|| -> eyre::Result<(Bookie, Booked)> {
323                let all_booked = corro_types::agent::BookedVersions::load_all_from_conn(&conn)
324                    .context("failed to load booked versions")?;
325
326                let bookie = Bookie::new(all_booked);
327                let booked = bookie.ensure(actor_id);
328                Ok((bookie, booked))
329            })?;
330
331        tracing::info!(elapsed = ?start.elapsed(), "Loaded booked versions");
332
333        Ok(Self {
334            pool,
335            bookie,
336            booked,
337            clock,
338            schema: Arc::new(schema),
339            actor_id,
340            cluster_id,
341        })
342    }
343}

345/// We currently only support updating the schema at startup
346fn update_schema(
347    conn: &mut WriteConn,
348    old_schema: Schema,
349    new_schema: Schema,
350) -> eyre::Result<Schema> {
351    // clone the previous schema and apply
352    let mut new_schema = {
353        let mut schema = old_schema.clone();
354        for (name, def) in new_schema.tables.iter() {
355            // overwrite table because users are expected to return a full table def
356            schema.tables.insert(name.clone(), def.clone());
357        }
358        schema
359    };
360
361    new_schema.constrain()?;
362
363    let tx = conn.immediate_transaction()?;
364
365    corro_types::schema::apply_schema(&tx, &old_schema, &mut new_schema)?;
366
367    for tbl_name in new_schema.tables.keys() {
368        tx.execute("DELETE FROM __corro_schema WHERE tbl_name = ?", [tbl_name])?;
369
370        let n = tx.execute("INSERT INTO __corro_schema SELECT tbl_name, type, name, sql, 'api' AS source FROM sqlite_schema WHERE tbl_name = ? AND type IN ('table', 'index') AND name IS NOT NULL AND sql IS NOT NULL", [tbl_name])?;
371        tracing::info!("Updated {n} rows in __corro_schema for table {tbl_name}");
372    }
373
374    tx.commit()?;
375    Ok(new_schema)
376}

378/// Verifies invariants and applies one-time configuration after schema migration.
379/// Failures are fatal — the database cannot be used safely.
380fn run_startup_checks(conn: &rusqlite::Connection, limits: Option<&DBLimits>) -> eyre::Result<()> {
381    use eyre::WrapErr;
382
383    conn.query_row("SELECT crsql_db_version()", [], |row| row.get::<_, i64>(0))
384        .context("cr-sqlite extension not loaded")?;
385
386    let mode: String = conn.query_row("PRAGMA journal_mode", [], |row| row.get(0))?;
387    eyre::ensure!(mode == "wal", "journal_mode is '{mode}', expected 'wal'");
388
389    let options: Vec<String> = {
390        let mut stmt = conn.prepare("PRAGMA compile_options")?;
391        stmt.query_map([], |row| row.get(0))?
392            .collect::<Result<_, _>>()?
393    };
394    eyre::ensure!(
395        !options.iter().any(|o| o == "THREADSAFE=0"),
396        "SQLite built with THREADSAFE=0; multi-threaded access is undefined behavior"
397    );
398
399    // mmap bypasses fsync — C extension writes go directly to the file without WAL protection
400    let mmap_size: i64 = conn.pragma_query_value(None, "mmap_size", |row| row.get(0))?;
401    if mmap_size != 0 {
402        tracing::warn!(
403            mmap_size,
404            "mmap enabled; C extension writes bypass fsync and can corrupt the database directly"
405        );
406    }
407
408    // catches corruption left by a previous crash before building on top of it
409    let check: String = conn.query_row("PRAGMA quick_check", [], |row| row.get(0))?;
410    eyre::ensure!(check == "ok", "database corruption detected: {check}");
411
412    // /tmp in Kubernetes pods is often a size-limited emptyDir; SQLITE_IOERR mid-query
413    // if it fills up
414    conn.pragma_update(None, "temp_store", "MEMORY")?;
415
416    // WAL mode is required; NORMAL skips the per-commit fsync that WAL FULL would add.
417    // Data can only be lost on a simultaneous OS crash + hardware failure, which is acceptable
418    // because corrosion nodes replicate state across peers.
419    conn.pragma_update(None, "synchronous", "NORMAL")?;
420
421    if let Some(limits) = limits {
422        if let Some(max_page_count) = limits.max_page_count {
423            conn.pragma_update(None, "max_page_count", max_page_count)?;
424            tracing::info!(max_page_count, "max_page_count limit applied");
425        }
426        if let Some(journal_size_limit) = limits.journal_size_limit {
427            conn.pragma_update(None, "journal_size_limit", journal_size_limit)?;
428            tracing::info!(journal_size_limit, "journal_size_limit applied");
429        }
430        if let Some(sync) = limits.synchronous {
431            conn.pragma_update(None, "synchronous", sync.as_str())?;
432            tracing::info!(?sync, "WAL synchronous override applied");
433        }
434        if let Some(n) = limits.wal_autocheckpoint {
435            conn.pragma_update(None, "wal_autocheckpoint", n)?;
436            tracing::info!(n, "wal_autocheckpoint set");
437        }
438    }
439
440    Ok(())
441}

443/// We keep a write-ahead-log, which under write-pressure can grow to multiple gigabytes and needs periodic truncation.
444///
445/// We don't want to schedule this task too often since it locks the whole DB.
446fn wal_checkpoint(conn: &rusqlite::Connection, timeout: Duration) -> eyre::Result<()> {
447    let start = std::time::Instant::now();
448
449    let orig: u64 = conn.pragma_query_value(None, "busy_timeout", |row| row.get(0))?;
450    conn.pragma_update(None, "busy_timeout", timeout.as_millis() as u64)?;
451
452    let busy: bool = conn.query_row("PRAGMA wal_checkpoint(TRUNCATE);", [], |row| row.get(0))?;
453    if busy {
454        tracing::warn!(?timeout, "WAL truncation incomplete: database still busy");
455    } else {
456        tracing::info!(elapsed = ?start.elapsed(), "WAL truncated");
457    }
458
459    drop(conn.pragma_update(None, "busy_timeout", orig));
460
461    Ok(())
462}

464async fn wal_checkpoint_over_threshold(
465    wal_path: &crate::Path,
466    pool: &SplitPool,
467    threshold: u64,
468) -> eyre::Result<bool> {
469    let wal_size = wal_path.metadata()?.len();
470    let should_truncate = wal_size > threshold;
471
472    if should_truncate {
473        let conn = if wal_size > (5 * threshold) {
474            tracing::warn!(
475                size = wal_size,
476                "wal_size is over 5x the threshold, trying to get a priority conn"
477            );
478            pool.write_priority().await?
479        } else {
480            pool.write_low().await?
481        };
482
483        fn calc_busy_timeout(wal_size: u64, threshold: u64) -> u64 {
484            let wal_size_gb = wal_size / (1024 * 1024 * 1024);
485            let threshold_gb = threshold / (1024 * 1024 * 1024);
486            let base_timeout = 30000;
487            if wal_size_gb <= threshold_gb {
488                return base_timeout;
489            }
490
491            // Double the timeout every 5gb and cap at 16 minutes
492            let diff = ((wal_size_gb - threshold_gb) / 5).min(5);
493            // add extra (five * diff) seconds for every extra 1gb over 4gb
494            let linear_increase = (wal_size_gb % 5) * 5000 * (diff + 1);
495            let timeout = base_timeout * 2u64.pow(diff as u32) + linear_increase;
496            // we are using a 16min timeout, something is wrong if we get here
497            if diff >= 5 {
498                tracing::warn!("WAL size is too large, setting busy timeout {timeout}ms");
499            }
500            timeout
501        }
502
503        let timeout = calc_busy_timeout(wal_size, threshold);
504        tokio::task::block_in_place(|| wal_checkpoint(&conn, Duration::from_millis(timeout)))?;
505    }
506
507    Ok(should_truncate)
508}

510/// Continuously runs an [incremental vacuum](https://sqlite.org/lang_vacuum.html) until the number of unused free pages is below the limit
511async fn vacuum_db(pool: &SplitPool, lim: u64) -> eyre::Result<()> {
512    let mut freelist: u64 = {
513        let conn = pool.read_readonly().await?;
514
515        let vacuum: u64 = conn.pragma_query_value(None, "auto_vacuum", |row| row.get(0))?;
516        eyre::ensure!(vacuum == 2, "auto_vacuum has to be set to INCREMENTAL");
517
518        conn.pragma_query_value(None, "freelist_count", |row| row.get(0))?
519    };
520
521    let cache_size: i64 = {
522        // update settings in write conn
523        let conn = pool.write_low().await?;
524        let cache_size = conn.pragma_query_value(None, "cache_size", |row| row.get(0))?;
525        conn.pragma_update(None, "cache_size", 100000)?;
526        cache_size
527    };
528
529    let result = async {
530        while freelist >= lim {
531            let conn = pool.write_low().await?;
532
533            tokio::task::block_in_place(|| {
534                let mut prepped = conn.prepare("pragma incremental_vacuum(1000)")?;
535                let mut rows = prepped.query([])?;
536
537                while let Ok(Some(_)) = rows.next() {}
538
539                freelist = conn.pragma_query_value(None, "freelist_count", |row| row.get(0))?;
540
541                Ok::<(), eyre::Error>(())
542            })?;
543
544            drop(conn);
545            tokio::time::sleep(Duration::from_secs(1)).await;
546        }
547
548        Ok(())
549    }
550    .await;
551
552    // restore the cache size even if vacuuming failed
553    let conn = pool.write_low().await?;
554    conn.pragma_update(None, "cache_size", cache_size)?;
555
556    result
557}

559/// Collects the `CRDT` bookkeeping left behind by deleted rows.
560///
561/// `cr-sqlite` keeps a sentinel row in `<table>__crsql_clock` for every row ever deleted,
562/// so that a delete can still win against a concurrent write from a peer that has yet to
563/// see it, and a matching entry in `<table>__crsql_pks`. Neither is ever removed: under
564/// row churn the base table stays flat while those two grow without bound, which is what
565/// eventually exhausts the volume. Once `retention` has passed every peer has long since
566/// converged and the tombstone is only taking up space.
567///
568/// Deletes at most `limit` tombstones, plus any primary keys orphaned by an earlier pass.
569/// Returns the `(tombstone, primary key)` row counts. The clock tables are internal to
570/// `cr-sqlite` and carry no triggers, so this does not itself produce a change to gossip.
571pub async fn reap_tombstones(
572    pool: &SplitPool,
573    table: &str,
574    retention: Duration,
575    clock: &crate::Clock,
576    limit: usize,
577) -> eyre::Result<(usize, usize)> {
578    let now = corro_types::broadcast::Timestamp::from(clock.0.new_timestamp()).to_ntp64();
579    let cutoff = corro_types::broadcast::Timestamp(now - uhlc::NTP64::from(retention));
580
581    let mut conn = pool.write_low().await?;
582
583    tokio::task::block_in_place(|| {
584        let tx = conn.immediate_transaction()?;
585
586        // col_name -1 marks the delete sentinel, and an even col_version means the row is
587        // currently deleted rather than resurrected by a later insert
588        let tombstones = tx
589            .prepare_cached(&format!(
590                "DELETE FROM {table}__crsql_clock WHERE key IN (
591                     SELECT key FROM {table}__crsql_clock
592                     WHERE col_name = -1 AND col_version % 2 = 0 AND ts < ?
593                     LIMIT ?
594                 )"
595            ))?
596            .execute(rusqlite::params![cutoff, limit])?;
597
598        // a key whose clock rows have all gone can never be referenced again
599        let pks = tx
600            .prepare_cached(&format!(
601                "DELETE FROM {table}__crsql_pks WHERE __crsql_key IN (
602                     SELECT __crsql_key FROM {table}__crsql_pks
603                     WHERE NOT EXISTS (
604                         SELECT 1 FROM {table}__crsql_clock WHERE key = __crsql_key
605                     )
606                     LIMIT ?
607                 )"
608            ))?
609            .execute(rusqlite::params![limit])?;
610
611        tx.commit()?;
612        Ok::<_, rusqlite::Error>((tombstones, pks))
613    })
614    .map_err(Into::into)
615}

617fn spawn_db_maintenance(
618    path: &crate::Path,
619    pool: SplitPool,
620    dbm: DBMaintenance,
621    clock: crate::Clock,
622    tables: Vec<String>,
623) {
624    let wal_path = {
625        let mut wp = path.to_path_buf();
626        wp.set_extension(format!("{}-wal", path.extension().unwrap_or_default()));
627        wp
628    };
629    let wal_threshold = dbm.wal_threshold as u64 * 1024 * 1024;
630
631    tokio::spawn(async move {
632        // try to initially truncate the WAL
633        match wal_checkpoint_over_threshold(wal_path.as_path(), &pool, wal_threshold).await {
634            Ok(truncated) if truncated => {
635                tracing::info!("initially truncated WAL");
636            }
637            Err(error) => {
638                tracing::error!(%error, "could not initially truncate WAL");
639            }
640            _ => {}
641        }
642
643        // large sleep right at the start to give node time to sync
644        tokio::time::sleep(Duration::from_secs(60)).await;
645
646        let mut vacuum_interval = tokio::time::interval(dbm.interval);
647
648        loop {
649            vacuum_interval.tick().await;
650            if let Err(error) = vacuum_db(&pool, dbm.max_free_pages).await {
651                tracing::error!(%error, "could not check freelist and vacuum");
652            }
653
654            if let Some(retention) = dbm.tombstone_retention {
655                for table in &tables {
656                    match reap_tombstones(&pool, table, retention, &clock, dbm.tombstone_batch)
657                        .await
658                    {
659                        Ok((0, 0)) => {}
660                        Ok((tombstones, pks)) => {
661                            tracing::debug!(table, tombstones, pks, "collected tombstones");
662                        }
663                        Err(error) => {
664                            tracing::error!(%error, table, "could not collect tombstones");
665                        }
666                    }
667                }
668            }
669
670            if let Err(error) =
671                wal_checkpoint_over_threshold(wal_path.as_path(), &pool, wal_threshold).await
672            {
673                tracing::error!(%error, "could not wal_checkpoint truncate");
674            }
675        }
676    });
677}

679#[cfg(test)]
680mod test {
681    use super::*;
682
683    /// The caps have to leave the volume with room to spare. `SQLITE_FULL` is only
684    /// recoverable while writes still land: `recover_space` checkpoints, vacuums and
685    /// purges rows, and every one of those needs to write.
686    #[test]
687    fn derived_limits_leave_recovery_headroom() {
688        const PAGE: u64 = 4096;
689
690        // 64 MiB is the volume the relays wedged on
691        for capacity in [64 << 20, 256 << 20, 1 << 30, 64u64 << 30] {
692            let (max_page_count, journal_size_limit) =
693                derive_limits(capacity, PAGE).expect("limits should be derivable");
694            let used = max_page_count * PAGE + journal_size_limit as u64;
695            let reserved = capacity.checked_sub(used).expect("caps exceed the volume");
696
697            assert!(
698                reserved >= capacity / 20,
699                "{capacity} byte volume left only {reserved} bytes for recovery"
700            );
701        }
702    }
703
704    /// An operator sizing the caps against their own volume has to win over the
705    /// derived ones, including when they set only one of the two.
706    #[test]
707    fn configured_limits_are_kept() {
708        let conn = rusqlite::Connection::open_in_memory().unwrap();
709        let mut limits = DBLimits {
710            max_page_count: Some(1234),
711            ..Default::default()
712        };
713
714        limits
715            .apply_volume_defaults(&conn, crate::Path::new("/"))
716            .unwrap();
717
718        assert_eq!(limits.max_page_count, Some(1234));
719        assert!(
720            limits.journal_size_limit.is_some(),
721            "the unset cap should still be derived"
722        );
723    }
