35/// The size of a single receive buffer, for a presumed MTU of 1500 rounded up
36/// to the next power of 2, which the io-uring buffer ring requires
37///
38/// A backend accepts this minus whatever metadata it keeps in the same buffer,
39/// see its `MAX_PACKET_LEN`. Larger datagrams are dropped.
40pub const RECV_BUFFER_LEN: usize = 2048;
41
42/// How often [`empty_packet_dropped`] may log
43const EMPTY_PACKET_LOG_INTERVAL: i64 = 5 * 60 * 1_000_000_000;
44
45/// When [`empty_packet_dropped`] last logged, as unix nanos
46static EMPTY_PACKET_LOGGED_AT: std::sync::atomic::AtomicI64 =
47    std::sync::atomic::AtomicI64::new(i64::MIN);
48
49/// Records an empty datagram being dropped
50///
51/// Counted every time, but logged at most once per [`EMPTY_PACKET_LOG_INTERVAL`],
52/// as a client that sends these sends them at whatever rate it ticks at.
53pub fn empty_packet_dropped() {
54    use crate::service::ALLOW_EMPTY_PACKETS_FLAG;
55
56    metrics::packets_dropped(metrics::READ, metrics::DropReason::EmptyPacket).inc();
57    crate::net::metrics::observe_dropped_packet(
58        metrics::READ,
59        "",
60        0,
61        metrics::DropReason::EmptyPacket,
62    );
63
64    let now = crate::time::UtcTimestamp::now().unix_nanos();
65
66    if claim_log_slot(&EMPTY_PACKET_LOGGED_AT, now, EMPTY_PACKET_LOG_INTERVAL) {
67        tracing::warn!(
68            "empty packets are dropped by default, enable --{ALLOW_EMPTY_PACKETS_FLAG} to allow them"
69        );
70    }
71}

73/// Whether `now` is at least `interval` past the timestamp in `last`, taking
74/// the slot for `now` when it is
75///
76/// Every worker reaches this for the same packet storm, so the slot is taken by
77/// a swap and only the thread that wins it logs.
78fn claim_log_slot(last: &std::sync::atomic::AtomicI64, now: i64, interval: i64) -> bool {
79    use std::sync::atomic::Ordering;
80
81    let previous = last.load(Ordering::Relaxed);
82
83    now.saturating_sub(previous) >= interval
84        && last
85            .compare_exchange(previous, now, Ordering::Relaxed, Ordering::Relaxed)
86            .is_ok()
87}

89/// Returns the canonical form of `ip` when it is a source we never forward, as
90/// a reply to a spoofed one of these would make the proxy a UDP amplifier
91///
92/// Dual stack sockets report ipv4 peers as ipv4-mapped ipv6, which matches none
93/// of the ipv6 predicates, so the mapping is undone before the check.
94pub fn disallowed_source_ip(ip: IpAddr) -> Option<IpAddr> {
95    let ip = ip.to_canonical();
96
97    let disallowed = match ip {
98        IpAddr::V4(v4) => {
99            v4.is_loopback() || v4.is_multicast() || v4.is_broadcast() || v4.is_unspecified()
100        }
101        IpAddr::V6(v6) => v6.is_loopback() || v6.is_multicast() || v6.is_unspecified(),
102    };
103
104    disallowed.then_some(ip)
105}

107/// Representation of an immutable set of bytes pulled from the network, this trait
108/// provides an abstraction over however the packet was received (epoll, io-uring, xdp)
109///
110/// Use [`PacketMut`] if you need a mutable representation.
111pub trait Packet: Sized {
112    /// Returns the underlying slice of bytes representing the packet.
113    fn as_slice(&self) -> &[u8];
114
115    /// Returns the size of the packet.
116    fn len(&self) -> usize;
117
118    /// Returns whether the given packet is empty.
119    fn is_empty(&self) -> bool {
120        self.len() == 0
121    }
122}
123
124/// Representation of an mutable set of bytes pulled from the network, this trait
125/// provides an abstraction over however the packet was received (epoll, io-uring, xdp)
126pub trait PacketMut: Sized + Packet {
127    /// Truncates the head by the specified number of bytes
128    fn remove_head(&mut self, length: usize);
129    /// Truncates the tail by the specified number of bytes
130    fn remove_tail(&mut self, length: usize);
131    /// Prepend the specified bytes
132    fn extend_head(&mut self, bytes: &[u8]);
133    /// Append the specified bytes
134    fn extend_tail(&mut self, bytes: &[u8]);
135    /// Returns an immutable version of the packet, this allows certain types
136    /// return a type that can be more cheaply cloned and shared.
137    fn freeze(self) -> bytes::Bytes;
138}
139
140impl PacketMut for bytes::BytesMut {
141    fn remove_head(&mut self, length: usize) {
142        drop(self.split_to(length));
143    }
144
145    fn extend_head(&mut self, bytes: &[u8]) {
146        let rest = self.split();
147        self.extend_from_slice(bytes);
148        self.unsplit(rest);
149    }
150
151    fn extend_tail(&mut self, bytes: &[u8]) {
152        self.extend_from_slice(bytes);
153    }
154
155    fn remove_tail(&mut self, length: usize) {
156        self.truncate(self.len().saturating_sub(length));
157    }
158
159    fn freeze(self) -> bytes::Bytes {
160        self.freeze()
161    }
162}
163
164impl Packet for bytes::BytesMut {
165    fn as_slice(&self) -> &[u8] {
166        self.as_ref()
167    }
168
169    fn is_empty(&self) -> bool {
170        self.is_empty()
171    }
172
173    fn len(&self) -> usize {
174        self.len()
175    }
176}

178/// Packet received from local port
179pub(crate) struct DownstreamPacket<'stack, P> {
180    pub(crate) contents: P,
181    pub(crate) source: SocketAddr,
182    pub(crate) filters: &'stack crate::filters::FilterChain,
183    pub(crate) allow_empty: bool,
184}
185
186impl<P: PacketMut> DownstreamPacket<'_, P> {
187    #[inline]
188    pub(crate) fn process<S: SessionManager>(
189        self,
190        worker_id: usize,
191        config: &Arc<Config>,
192        sessions: &S,
193        destinations: &mut Vec<crate::net::Destination>,
194    ) {
195        let payload_size = self.contents.len();
196        tracing::trace!(
197            id = worker_id,
198            size = payload_size,
199            source = %self.source,
200            "received packet from downstream"
201        );
202
203        let timer = metrics::processing_time(metrics::READ).start_timer();
204        if let Err(error) = self.process_inner(config, sessions, destinations) {
205            error.inc_system_errors_total(metrics::READ, &metrics::EMPTY);
206            metrics::packets_dropped_total(
207                metrics::READ,
208                error.drop_reason(),
209                error.filter_name(),
210                "",
211            )
212            .inc();
213            crate::net::metrics::observe_dropped_packet_with_filter(
214                metrics::READ,
215                "",
216                payload_size,
217                error.drop_reason(),
218                error.filter_name(),
219            );
220        }
221
222        timer.stop_and_record();
223    }
224
225    /// Processes a packet by running it through the filter chain.
226    #[inline]
227    fn process_inner<S: SessionManager>(
228        self,
229        config: &Arc<Config>,
230        sessions: &S,
231        destinations: &mut Vec<crate::net::Destination>,
232    ) -> Result<(), PipelineError> {
233        if self.contents.is_empty() && !self.allow_empty {
234            empty_packet_dropped();
235            return Ok(());
236        }
237
238        let Some(clusters) = config
239            .dyn_cfg
240            .clusters()
241            .filter(|c| c.read().has_endpoints())
242        else {
243            tracing::trace!("no upstream endpoints");
244            return Err(PipelineError::NoUpstreamEndpoints);
245        };
246
247        #[cfg(not(debug_assertions))]
248        if let Some(ip) = disallowed_source_ip(self.source.ip()) {
249            return Err(PipelineError::DisallowedSourceIP(ip));
250        }
251
252        let cm = clusters.clone_value();
253        let mut context = ReadContext::new(&cm, self.source.into(), self.contents, destinations);
254        self.filters
255            .read(&mut context)
256            .map_err(PipelineError::Filter)?;
257
258        let ReadContext { contents, .. } = context;
259
260        if destinations.is_empty() {
261            return Ok(());
262        }
263
264        // Similar to bytes::BytesMut::freeze, we turn the mutable pool buffer
265        // into an immutable one with its own internal arc so it can be cloned
266        // cheaply and returned to the pool once all references are dropped
267        let contents = contents.freeze();
268
269        // A single destination is the most likely outcome for a typical workload
270        // so we can avoid an unnecessary buffer clone
271        if destinations.len() == 1
272            && let Some(dest) = destinations.pop()
273        {
274            let session_key = SessionKey {
275                source: self.source,
276                dest: dest.address.to_socket_addr()?,
277            };
278
279            sessions.send(session_key, contents, dest.cluster)?;
280        } else {
281            for dest in destinations.drain(0..) {
282                let session_key = SessionKey {
283                    source: self.source,
284                    dest: dest.address.to_socket_addr()?,
285                };
286
287                sessions.send(session_key, contents.clone(), dest.cluster)?;
288            }
289        }
290
291        Ok(())
292    }
293}

295/// Runs a packet through the filter chain with a no-op session manager.
296/// Used by benchmarks to measure pipeline overhead without any IO.
297pub fn bench_process_packet(
298    payload: bytes::BytesMut,
299    source: std::net::SocketAddr,
300    config: &Arc<Config>,
301    filter_chain: &crate::filters::FilterChain,
302    destinations: &mut Vec<crate::net::Destination>,
303) {
304    struct Noop;
305    impl crate::net::sessions::SessionManager for Noop {
306        fn send(
307            &self,
308            _key: crate::net::sessions::SessionKey,
309            _contents: bytes::Bytes,
310            _cluster: Option<crate::net::endpoint::Locality>,
311        ) -> Result<(), crate::net::PipelineError> {
312            Ok(())
313        }
314    }
315    DownstreamPacket {
316        contents: payload,
317        source,
318        filters: filter_chain,
319        allow_empty: true,
320    }
321    .process(0, config, &Noop, destinations);
322}

324/// Spawns a background task that sits in a loop, receiving packets from the passed in socket.
325/// Each received packet is placed on a queue to be processed by a worker task.
326/// This function also spawns the set of worker tasks responsible for consuming packets
327/// off the aforementioned queue and processing them through the filter chain and session
328/// pipeline.
329pub fn spawn_receivers(
330    config: Arc<Config>,
331    socket: socket2::Socket,
332    worker_sends: Vec<crate::net::PacketQueue>,
333    sessions: &Arc<SessionPool>,
334    backend: crate::net::io::UdpBackend,
335    recv_ring_len: u16,
336    allow_empty_packets: bool,
337) -> crate::Result<()> {
338    let port = crate::net::socket_port(&socket);
339
340    let Some(pfc) = config.dyn_cfg.cached_filter_chain() else {
341        eyre::bail!("the ProxyFilterChain state was not configured");
342    };
343
344    for (worker_id, ws) in worker_sends.into_iter().enumerate() {
345        let worker = crate::net::io::Listener {
346            worker_id,
347            port,
348            config: config.clone(),
349            sessions: sessions.clone(),
350            backend,
351            allow_empty_packets,
352        };
353
354        worker.spawn_io_loop(ws, pfc.clone(), recv_ring_len)?;
355    }
356
357    Ok(())
358}

361mod tests {
362    use std::net::{Ipv4Addr, Ipv6Addr};
363
364    use super::*;

366    /// A dual stack socket reports an ipv4 peer in ipv4-mapped form, so every
367    /// ipv4 source has to be caught in both spellings
368    #[test]
369    fn disallows_amplification_sources() {
370        for v4 in [
371            Ipv4Addr::LOCALHOST,
372            Ipv4Addr::BROADCAST,
373            Ipv4Addr::UNSPECIFIED,
374            // multicast = 224.0.0.0/4
375            Ipv4Addr::new(224, 0, 0, 0),
376            Ipv4Addr::new(239, 255, 255, 255),
377        ] {
378            let canonical = Some(IpAddr::V4(v4));
379            assert_eq!(disallowed_source_ip(v4.into()), canonical, "{v4}");
380            assert_eq!(
381                disallowed_source_ip(v4.to_ipv6_mapped().into()),
382                canonical,
383                "::ffff:{v4}"
384            );
385        }
386
387        for v6 in [
388            Ipv6Addr::LOCALHOST,
389            Ipv6Addr::UNSPECIFIED,
390            // multicast = any address starting with 0xff
391            Ipv6Addr::new(0xff00, 0, 0, 0, 0, 0, 0, 0),
392        ] {
393            assert_eq!(
394                disallowed_source_ip(v6.into()),
395                Some(IpAddr::V6(v6)),
396                "{v6}"
397            );
398        }
399
400        // Ordinary clients, in every spelling they can arrive in
401        let routable = Ipv4Addr::new(203, 0, 113, 1);
402        for allowed in [
403            IpAddr::V4(routable),
404            IpAddr::V6(routable.to_ipv6_mapped()),
405            IpAddr::V6(Ipv6Addr::new(0x2001, 0xdb8, 0, 0, 0, 0, 0, 1)),
406        ] {
407            assert_eq!(disallowed_source_ip(allowed), None, "{allowed}");
408        }
409    }
