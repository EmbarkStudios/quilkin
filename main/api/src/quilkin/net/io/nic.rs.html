6/// Different scheduling policies that can be used for XDP worker threads
7///
8/// Note there are other scheduling policies, but these are the only ones relevant for quilkin
9#[derive(Copy, Clone, Default)]
10#[repr(i32)]
11pub enum ThreadPolicy {
12    /// `SCHED_OTHER` default scheduling policy in Linux
13    #[default]
14    Default = 0,
15    /// [`SCHED_FIFO`](https://man.archlinux.org/man/sched.7.en#SCHED_FIFO:_First_in-first_out_scheduling) real time scheduling policy
16    Fifo = 1,
17    /// [`SCHED_RR`](https://man.archlinux.org/man/sched.7.en#SCHED_RR:_Round-robin_scheduling), FIFO but with time slicing
18    RoundRobin = 2,
19}
20
21impl fmt::Debug for ThreadPolicy {
22    fn fmt(&self, f: &mut fmt::Formatter<'_>) -> fmt::Result {
23        write!(f, "{self}")
24    }
25}
26
27impl fmt::Display for ThreadPolicy {
28    fn fmt(&self, f: &mut fmt::Formatter<'_>) -> fmt::Result {
29        match self {
30            Self::Default => f.write_str("SCHED_OTHER"),
31            Self::Fifo => f.write_str("SCHED_FIFO"),
32            Self::RoundRobin => f.write_str("SCHED_RR"),
33        }
34    }
35}
36
37impl clap::ValueEnum for ThreadPolicy {
38    fn value_variants<'a>() -> &'a [Self] {
39        &[Self::Default, Self::Fifo, Self::RoundRobin]
40    }
41
42    fn to_possible_value(&self) -> Option<clap::builder::PossibleValue> {
43        use clap::builder::PossibleValue as pv;
44        Some(match self {
45            Self::Default => pv::new("SCHED_OTHER"),
46            Self::Fifo => pv::new("SCHED_FIFO"),
47            Self::RoundRobin => pv::new("SCHED_RR"),
48        })
49    }
50}
