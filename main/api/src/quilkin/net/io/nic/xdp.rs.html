59    pub require_tx_checksum: bool,
60    /// Total amount of packets per UMEM, which is used to determine the size of each ring
61    pub packets_per_queue: u32,
62    /// Worker thread scheduling configuration
63    pub worker_thread_scheduling: WorkerThreadScheduling,
64}

145/// Attempts to setup XDP by querying NIC support and allocating ring buffers
146/// based on user configuration, failing if requirements cannot be met
147///
148/// This function currently only supports one mode of operation, which is that
149/// Returns `true` if a unique default NIC suitable for XDP is present.
150///
151/// This mirrors the `NicConfig::Default` selection in [`setup_xdp_io`] but
152/// stops before allocating sockets or loading the eBPF program.
153pub fn is_available() -> bool {
154    let Ok(iter) = xdp::nic::InterfaceIter::new() else {
155        return false;
156    };
157    let mut chosen = None;
158    for iface in iter {
159        if let Some(c) = chosen {
160            if iface != c {
161                return false;
162            }
163        } else {
164            chosen = Some(iface);
165        }
166    }
167    chosen.is_some()
168}

170/// Some drivers (eg `gve`) reserve queues for XDP, rejecting attach at the
171/// full count. Returns the next smaller count to retry, or `None` if there's
172/// no room left to shrink.
173fn next_queue_count_to_try(current: u32) -> Option<u32> {
174    let half = current / 2;
175    (half >= 1).then_some(half)
176}

178/// Loads and attaches `ebpf_prog` to `nic`, starting at `initial_channels`'
179/// current queue count and only shrinking it reactively on an attach failure
180/// — NICs that don't need it are never touched. Returns the link and the
181/// queue count XDP ended up using.
182fn attach_xdp_program(
183    ebpf_prog: &mut quilkin_xdp::EbpfProgram,
184    nic: NicIndex,
185    queues: &mut quilkin_xdp::xdp::nic::Queues,
186) -> Result<quilkin_xdp::aya::programs::xdp::XdpLinkId, quilkin_xdp::aya::programs::ProgramError> {
187    ebpf_prog.load_into_kernel()?;
188
189    let mode = quilkin_xdp::aya::programs::xdp::XdpMode::default();
190    // We've already synchronized the tx and rx counts if needed before this function
191    let mut queue_count = queues.rx_count();
192
193    loop {
194        match ebpf_prog.attach(nic, mode) {
195            Ok(link) => return Ok(link),
196            Err(error) => {
197                let reduced = queue_count / 2;
198                if reduced < 1 {
199                    return Err(error);
200                }
201
202                queues.set_rx_and_tx(reduced);
203
204                if let Err(shrink_error) = nic.set_queue_counts(queues) {
205                    tracing::warn!(
206                        %shrink_error,
207                        ?nic,
208                        "failed to reduce NIC queue count for XDP, giving up"
209                    );
210                    return Err(error);
211                }
212
213                tracing::warn!(
214                    %error,
215                    ?nic,
216                    from = queue_count,
217                    to = reduced,
218                    "XDP attach rejected at current queue count, retrying with fewer queues"
219                );
220                queue_count = reduced;
221            }
222        }
223    }
224}

226/// Binds an `AF_XDP` socket to every available queue on the NIC, and when [`spawn`]
227/// is invoked, each socket is processed in its own thread
228pub fn setup_xdp_io(config: XdpConfig<'_>) -> Result<XdpWorkers, XdpSetupError> {
229    // This is validated when parsing arguments, but sanity check just in case eg. tests
230    assert!(
231        config.packets_per_queue.is_power_of_two(),
232        "packets per queue was not a power of 2"
233    );
234
235    let nic = match config.nic {
236        NicConfig::Default => {
237            let mut chosen = None;
238
239            for iface in xdp::nic::InterfaceIter::new().map_err(NicUnavailable::Query)? {
240                if let Some(chosen) = chosen {
241                    if iface != chosen {
242                        return Err(NicUnavailable::NoAvailableDefault.into());
243                    }
244                } else {
245                    chosen = Some(iface);
246                }
247            }
248
249            chosen.ok_or(NicUnavailable::NoAvailableDefault)?
250        }
251        NicConfig::Name(name) => {
252            let cname = std::ffi::CString::new(name)
253                .map_err(|_nul| NicUnavailable::UnknownName(name.to_owned()))?;
254            xdp::nic::NicIndex::lookup_by_name(&cname)
255                .map_err(NicUnavailable::Query)?
256                .ok_or_else(|| NicUnavailable::UnknownName(name.to_owned()))?
257        }
258        NicConfig::Index(index) => xdp::nic::NicIndex::new(index),
259    };
260
261    let name = nic
262        .name()
263        .map_err(|_err| NicUnavailable::UnknownIndex(nic.into()))?;
264
265    tracing::info!(nic = ?nic, "selected NIC");
266
267    let mut device_caps = nic
268        .query_capabilities()
269        .map_err(|err| XdpSetupError::NicQuery(name, err))?;
270
271    tracing::info!(?device_caps, nic = ?nic, "XDP features for device");
272
273    if config.require_zero_copy
274        && matches!(device_caps.zero_copy, xdp::nic::XdpZeroCopy::Unavailable)
275    {
276        tracing::error!(?device_caps, nic = ?nic, "XDP features for device");
277        return Err(XdpSetupError::ZeroCopyUnavailable(name));
278    }
279
280    if config.require_tx_checksum && !device_caps.tx_metadata.checksum() {
281        tracing::error!(?device_caps, nic = ?nic, "XDP features for device");
282        return Err(XdpSetupError::TxChecksumUnavailable(name));
283    }
284
285    let (ipv4, ipv6) = nic
286        .addresses()
287        .and_then(|(ipv4, ipv6)| {
288            if ipv4.is_none() && ipv6.is_none() {
289                Err(std::io::Error::new(
290                    std::io::ErrorKind::AddrNotAvailable,
291                    "neither an ipv4 nor ipv6 address could be determined for the device",
292                ))
293            } else {
294                Ok((
295                    ipv4.unwrap_or(std::net::Ipv4Addr::new(0, 0, 0, 0)),
296                    ipv6.unwrap_or(std::net::Ipv6Addr::from_bits(0)),
297                ))
298            }
299        })
300        .map_err(|err| XdpSetupError::AddressQuery(name, err))?;
301
302    // We don't support unaligned chunks, so this size can only be 2k or 4k,
303    // and we only need 2k since we only care about non-fragmented UDP packets
304    const PACKET_SIZE: u64 = 2 * 1024;
305
306    // It's extremely unlikely these number will ever differ, but if they do we take the minimum
307    let rx_count = device_caps.queues.rx_count();
308    let tx_count = device_caps.queues.tx_count();
309
310    if rx_count != tx_count {
311        tracing::warn!(
312            ?nic,
313            rx_count,
314            tx_count,
315            "the selected NIC has a different number of RX and TX queues"
316        );
317        let qc = rx_count.min(tx_count);
318
319        // If the tx count is higher it doesn't really matter since they just wouldn't be used, but if we have more rx than tx
320        // we need to reduce as otherwise we might not be able to send on some of the i/o loops
321        if rx_count > tx_count {
322            device_caps.queues.set_rx_and_tx(qc);
323            if let Err(error) = nic.set_queue_counts(&device_caps.queues) {
324                tracing::error!(%error, ?nic, rx_count, tx_count, "failed to synchronize queue counts before attach");
325                return Err(XdpSetupError::NicQuery(name, error));
326            }
327        }
328    }
329
330    let mut ebpf_prog = quilkin_xdp::EbpfProgram::load(config.external_port, config.qcmp_port)?;
331
332    // Attach before binding: some drivers (eg gve) need XDP already enabled
333    // before a socket can bind a queue with `XDP_ZEROCOPY`.
334    let xdp_link = attach_xdp_program(&mut ebpf_prog, nic, &mut device_caps.queues)?;
335
336    // In GCP, and possibly (probably?) other virtualized environments, the NIC will report the full theoretical speed
337    // even if in reality the amount of bandwidth will be shared among multiple tenants, maybe even other instances of
338    // this proxy, so rather than try to account for the amount of variation between clouds/virtualized/physical capabilities,
339    // we just require the operator to specify the number of packets they want to support
340
341    let umem_cfg = xdp::umem::UmemCfgBuilder {
342        frame_size: xdp::umem::FrameSize::TwoK,
343        // Provide enough headroom so that we can convert an ipv4 header to ipv6
344        // header without needing to copy any bytes. note this doesn't take into
345        // account if a filter adds or removes bytes from the beginning of the
346        // data payload
347        head_room: (xdp::packet::net_types::Ipv6Hdr::LEN - xdp::packet::net_types::Ipv4Hdr::LEN)
348            as u32,
349        frame_count: config.packets_per_queue,
350        // TODO: This should be done in the type system so we can avoid logic
351        // that doesn't change during the course of operation, but for now just
352        // do it at runtime
353        tx_checksum: device_caps.tx_metadata.checksum(),
354        ..Default::default()
355    }
356    .build()?;
357
358    let ring_size = config.packets_per_queue >> 2;
359
360    let ring_cfg = xdp::RingConfigBuilder {
361        rx_count: ring_size,
362        tx_count: ring_size,
363        fill_count: ring_size,
364        completion_count: ring_size,
365    }
366    .build()?;
367    let ring_heap = ring_cfg.heap_size();
368    let workers =
369        ebpf_prog.create_and_bind_sockets(nic, umem_cfg, &device_caps, ring_cfg, ring_size)?;
370
371    // Report the amount of heap memory we're using specifically for XDP
372    fn byte_units(b: u64) -> String {
373        let mut units = b as f64;
374        let mut unit = 0;
375        const UNITS: &[&str] = &["B", "KiB", "MiB", "GiB"];
376
377        while units > 1024.0 && unit + 1 < UNITS.len() {
378            units /= 1024.0;
379            unit += 1;
380        }
381
382        format!("{units:.02}{}", UNITS[unit])
383    }
384
385    let total_heap_memory = byte_units(
386        (
387            // The amount of memory mapped for each of the rings shared between the kernel and userspace
388            ring_heap as u64
389            // The amount of memory mapped in each Umem
390            + config.packets_per_queue as u64 * PACKET_SIZE
391            // The amount of memory allocated for the RX and TX slabs
392            + ring_size as u64 * std::mem::size_of::<xdp::Packet>() as u64
393        ) * workers.len() as u64,
394    );
395    tracing::info!(
396        queue_count = workers.len(),
397        total_heap_memory,
398        "total heap memory used by XDP workers"
399    );
400
401    let mut worker_thread_scheduling = config.worker_thread_scheduling;
402
403    // SAFETY: We restrict the policy to only a select few, and still check the return value for an error
404    unsafe {
405        let min = libc::sched_get_priority_min(worker_thread_scheduling.thread_policy as _);
406        let max = libc::sched_get_priority_max(worker_thread_scheduling.thread_policy as _);
407
408        if min < 0 || max < 0 {
409            return Err(XdpSetupError::ThreadPriorityClamp(
410                worker_thread_scheduling.thread_policy,
411                std::io::Error::last_os_error(),
412            ));
413        }
414
415        worker_thread_scheduling.thread_priority =
416            worker_thread_scheduling.thread_priority.clamp(min, max);
417    }
418
419    Ok(XdpWorkers {
420        ebpf_prog,
421        workers,
422        nic,
423        xdp_link,
424        external_port: config.external_port.into(),
425        qcmp_port: config.qcmp_port.into(),
426        ipv4,
427        ipv6,
428        worker_thread_scheduling,
429    })
430}

432pub struct XdpLoop {
433    threads: Vec<std::thread::JoinHandle<()>>,
434    ebpf_prog: quilkin_xdp::EbpfProgram,
435    xdp_link: quilkin_xdp::aya::programs::xdp::XdpLinkId,
436    shutdown: Arc<std::sync::atomic::AtomicBool>,
437}
438
439impl XdpLoop {
440    /// Detaches the eBPF program from the attacked NIC and cancels all I/O
441    /// threads, waiting for them to exit
442    pub fn shutdown(mut self, wait: bool) {
443        if let Err(error) = self.ebpf_prog.detach(self.xdp_link) {
444            tracing::error!(%error, "failed to detach eBPF program");
445        }
446
447        self.shutdown
448            .store(true, std::sync::atomic::Ordering::Relaxed);
449
450        if !wait {
451            return;
452        }
453
454        tracing::info!("waiting on XDP workers");
455        let start = std::time::Instant::now();
456
457        for jh in self.threads {
458            if let Err(error) = jh.join() {
459                if let Some(error) = error.downcast_ref::<&'static str>() {
460                    tracing::error!(error, "XDP I/O thread encountered error");
461                } else if let Some(error) = error.downcast_ref::<String>() {
462                    tracing::error!(error, "XDP I/O thread encountered error");
463                } else {
464                    tracing::error!(?error, "XDP I/O thread encountered error");
465                };
466            }
467        }
468
469        tracing::info!(elapsed = ?start.elapsed(), "finished shutting down XDP workers");
470    }
471}

473fn spawn_worker<F, T>(
474    i: usize,
475    scheduling: WorkerThreadScheduling,
476    f: F,
477) -> std::io::Result<std::thread::JoinHandle<T>>
478where
479    F: FnOnce() -> T,
480    F: Send + 'static,
481    T: Send + 'static,
482{
483    std::thread::Builder::new()
484        .name(format!("xdp-io-{i}"))
485        .spawn(move || {
486            'ts: {
487                // SAFETY: We've already validated the thread priority is clamped between the valid range and all other arguments
488                // are valid when each syscall is invoked
489                unsafe {
490                    if libc::sched_setscheduler(0, scheduling.thread_policy as _, &libc::sched_param {
491                        sched_priority: scheduling.thread_priority,
492                    }) != 0 {
493                        tracing::warn!(thread = std::thread::current().name(), error = %std::io::Error::last_os_error(), "failed to set XDP thread scheduler");
494                    }
495
496                    if !scheduling.pin_threads {
497                        break 'ts;
498                    }
499
500                    let mut set: libc::cpu_set_t = std::mem::zeroed();
501                    libc::CPU_SET(i, &mut set);
502                    if libc::pthread_setaffinity_np(libc::pthread_self(), std::mem::size_of::<libc::cpu_set_t>(), &set) != 0 {
503                        tracing::warn!(thread = std::thread::current().name(), error = %std::io::Error::last_os_error(), "failed to pin XDP worker to core");
504                    }
505                }
506            }
507
508            f()
509        })
510}

512/// The entrypoint into the XDP I/O loop.
513///
514/// This spawns a named thread for each configured XDP socket to run the packet
515/// receiving + processing + sending. The eBPF program used to route packets to
516/// the XDP sockets is already attached to the NIC by [`setup_xdp_io`] by this
517/// point.
518///
519/// # Errors
520///
521/// This can fail if threads can not be spawned for some reason (unlikely)
522pub fn spawn(workers: XdpWorkers, config: process::ConfigState) -> Result<XdpLoop, XdpSpawnError> {
523    let nic = workers.nic;
524    let ebpf_prog = workers.ebpf_prog;
525    let xdp_link = workers.xdp_link;
526    let external_port = workers.external_port;
527    let qcmp_port = workers.qcmp_port;
528    let ipv4 = workers.ipv4;
529    let ipv6 = workers.ipv6;
530    let session_state = Arc::new(process::SessionState::default());
531    let shutdown = Arc::new(std::sync::atomic::AtomicBool::new(false));
532
533    let queue_count = workers.workers.len();
534    let mut threads = Vec::with_capacity(queue_count);
535    for (i, mut worker) in workers.workers.into_iter().enumerate() {
536        let cfg = config.clone();
537        let ss = session_state.clone();
538        let shutdown = shutdown.clone();
539
540        let jh = spawn_worker(i, workers.worker_thread_scheduling, move || {
541            // Enqueue buffers to the fill ring to ensure that we don't miss any packets
542            // SAFETY: we keep the umem alive for as long as the socket is alive
543            unsafe {
544                if let Err(error) =
545                    worker
546                        .fill
547                        .enqueue(&mut worker.umem, worker.ring_len >> 2, true)
548                {
549                    tracing::error!(%error, "failed to kick fill ring during initial spinup");
550                }
551            };
552
553            io_loop(
554                worker,
555                external_port,
556                qcmp_port,
557                cfg,
558                ss,
559                ipv4,
560                ipv6,
561                shutdown.clone(),
562            );
563        })
564        .map_err(XdpSpawnError::Thread)?;
565
566        threads.push(jh);
567    }
568
569    tracing::info!(?nic, queue_count, "XDP is running");
570
571    Ok(XdpLoop {
572        threads,
573        ebpf_prog,
574        xdp_link,
575        shutdown,
576    })
577}

583/// The core I/O loop
584///
585/// All of the ring operations are done in this loop so that the actual
586/// [`process::process_packets`] code can be cleanly tested without relying on
587/// a fully setup XDP socket/rings, relying only on a `Umem` (memory map)
588#[allow(clippy::too_many_arguments)]
589fn io_loop(
590    worker: quilkin_xdp::XdpWorker,
591    external_port: NetworkU16,
592    qcmp_port: NetworkU16,
593    mut config: process::ConfigState,
594    sessions: Arc<process::SessionState>,
595    local_ipv4: std::net::Ipv4Addr,
596    local_ipv6: std::net::Ipv6Addr,
597    shutdown: Arc<std::sync::atomic::AtomicBool>,
598) {
599    let quilkin_xdp::XdpWorker {
600        mut umem,
601        socket,
602        mut fill,
603        mut rx,
604        mut tx,
605        mut completion,
606        ring_len,
607    } = worker;
608
609    const POLL_TIMEOUT: xdp::socket::PollTimeout =
610        xdp::socket::PollTimeout::new(Some(std::time::Duration::from_millis(100)));
611
612    let mut state = process::State {
613        external_port,
614        qcmp_port,
615        destinations: Vec::with_capacity(1),
616        addr_to_asn: Default::default(),
617        sessions,
618        local_ipv4,
619        local_ipv6,
620        last_receive: UtcTimestamp::now(),
621    };
622
623    use xdp::slab::{HeapSlab, Slab};
624
625    let mut rx_slab = HeapSlab::with_capacity(ring_len);
626    let mut tx_slab = HeapSlab::with_capacity(ring_len);
627    let mut pending_sends = 0;
628    let mut outstanding = umem.outstanding() as i64;
629
630    crate::metrics::allocated_xdp_packets().add(outstanding);
631
632    // SAFETY: the cases of unsafe in this code block all concern the relationship
633    // between frames and the Umem, the frames cannot outlive the Umem which is
634    // the owner of the actual memory map
635    unsafe {
636        while !shutdown.load(std::sync::atomic::Ordering::Relaxed) {
637            // Wait for packets to be received, note that
638            // [poll](https://www.man7.org/linux/man-pages/man2/poll.2.html) also acts
639            // as a [cancellation point](https://www.man7.org/linux/man-pages/man7/pthreads.7.html),
640            // so shutdown will cause the thread to exit here
641            let Ok(true) = socket.poll_read(POLL_TIMEOUT) else {
642                continue;
643            };
644
645            let recvd = rx.recv(&umem, &mut rx_slab);
646
647            // Ensure the fill ring doesn't get starved, which could drop packets
648            if let Err(error) = fill.enqueue(&mut umem, ring_len - recvd, true) {
649                // EAGAIN means the wakeup wasn't delivered, but the buffers are
650                // already enqueued in the ring, so the kernel will pick them up
651                // on the next successful wakeup or its own polling; not an error.
652                if error.raw_os_error() != Some(libc::EAGAIN) {
653                    // This is shoehorning an error that isn't attributable to a particular packet
654                    crate::metrics::errors_total(
655                        crate::metrics::Direction::Read,
656                        &io_error_to_discriminant(error),
657                        &crate::metrics::EMPTY,
658                    )
659                    .inc();
660                }
661            }
662
663            // Process each of the packets that we received, potentially queuing
664            // packets to be sent
665            process::process_packets(
666                &mut rx_slab,
667                &mut umem,
668                &mut tx_slab,
669                &mut config,
670                &mut state,
671            );
672
673            let before = tx_slab.len();
674            let enqueued_sends = match tx.send(&mut tx_slab, true) {
675                Ok(es) => es,
676                Err(error) => {
677                    // EAGAIN means the wakeup wasn't delivered, but the packets are
678                    // already enqueued in the ring, so the kernel will send them on
679                    // the next successful wakeup or its own polling; not an error.
680                    if error.raw_os_error() != Some(libc::EAGAIN) {
681                        // This is shoehorning an error that isn't attributable to a particular packet
682                        crate::metrics::errors_total(
683                            crate::metrics::Direction::Read,
684                            &io_error_to_discriminant(error),
685                            &crate::metrics::EMPTY,
686                        )
687                        .inc();
688                    }
689
690                    before - tx_slab.len()
691                }
692            };
693
694            // Return frames that have completed sending
695            pending_sends += enqueued_sends;
696            pending_sends -= completion.dequeue(&mut umem, pending_sends);
697
698            let new = umem.outstanding() as i64;
699            crate::metrics::allocated_xdp_packets().add(new - outstanding);
700            outstanding = new;
701        }
702    }
703}
704
705#[inline]
706fn io_error_to_discriminant(error: std::io::Error) -> std::borrow::Cow<'static, str> {
707    let Some(code) = error.raw_os_error() else {
708        return error.to_string().into();
709    };
710
711    match code {
712        libc::EBUSY => "EBUSY".into(),
713        libc::ENOBUFS => "ENOBUFS".into(),
714        libc::EAGAIN => "EAGAIN".into(),
715        libc::ENETDOWN => "ENETDOWN".into(),
716        other => format!("{other}").into(),
717    }
718}
